    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   410→  const processRecipients = (value: string): string => {

   421→        // If it contains spaces or looks like a name, return as-is (will fail validation)
   422→        return r
   423→      })

   507→  const handleSend = async (scheduledAtDate?: Date | null) => {
   508→    // Prevent double-send from double-click or rapid taps
   509→    if (isSendingRef.current) return
   510→    isSendingRef.current = true
   511→
   512→    try {
   513→      const e: Record<string, string> = {}
   514→      if (!to.trim()) e.to = 'Recipient is required'
   515→      if (!subject.trim()) e.subject = 'Subject is required'
   516→      const text = editor?.getText().trim()
   517→      if (!text || text.length === 0) e.body = 'Message body is required'
   518→      // Validate CC recipients if provided
   519→      if (showCc && cc.trim()) {
   520→        const ccParts = cc.split(',').map(r => r.trim()).filter(Boolean)
   521→        for (const part of ccParts) {
   522→          const processed = processRecipients(part)
   523→          if (!processed.endsWith('@ezy.af')) {
   524→            e.cc = `"${part}" is not a valid @ezy.af address`
   525→            break
   526→          }
   527→        }
   528→      }
   529→      // Validate BCC recipients if provided
   530→      if (showBcc && bcc.trim()) {
   531→        const bccParts = bcc.split(',').map(r => r.trim()).filter(Boolean)
   532→        for (const part of bccParts) {
   533→          const processed = processRecipients(part)
   534→          if (!processed.endsWith('@ezy.af')) {
   535→            e.bcc = `"${part}" is not a valid @ezy.af address`
   536→            break
   537→          }
   538→        }
   539→      }
   540→      setErrors(e)
   541→      if (Object.keys(e).length > 0) {
   542→        isSendingRef.current = false
   543→        return
   544→      }
   545→
   546→      const html = editor?.getHTML() || ''
   547→      const currentAttachments = attachments
   548→
   549→      // Upload attachments FIRST
   550→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined
   551→      if (currentAttachments.length > 0) {
   552→        setIsUploading(true)
   553→        setUploadProgress(0)
