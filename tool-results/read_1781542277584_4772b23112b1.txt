    14→
    15→const MONTHS = [
    16→  'January', 'February', 'March', 'April', 'May', 'June',
    17→  'July', 'August', 'September', 'October', 'November', 'December'
    18→]
    19→
    20→function generateDays(month: number, year: number) {
    21→  if (!month || !year) return Array.from({ length: 31 }, (_, i) => i + 1)
    22→  const daysInMonth = new Date(year, month, 0).getDate()
    23→  return Array.from({ length: daysInMonth }, (_, i) => i + 1)
    24→}
    25→
    26→function generateYears() {
    27→  const currentYear = new Date().getFullYear()
    28→  return Array.from({ length: 100 }, (_, i) => currentYear - 13 - i)
    29→}
    30→
    31→function getPasswordStrength(password: string) {
    32→  if (!password) return { score: 0, label: '', color: '' }
    33→  let score = 0
    34→  if (password.length >= 8) score++
    35→  if (/[a-z]/.test(password)) score++
    36→  if (/[A-Z]/.test(password)) score++
    37→  if (/[0-9]/.test(password)) score++
    38→  if (/[^a-zA-Z0-9]/.test(password)) score++
    39→  if (score <= 2) return { score, label: 'Weak', color: 'bg-red-500' }
    40→  if (score <= 3) return { score, label: 'Medium', color: 'bg-amber-500' }
    41→  return { score, label: 'Strong', color: 'bg-emerald-500' }
    42→}
    43→
    44→const nameSchema = z.object({
    45→  firstName: z.string().min(2, 'First name must be at least 2 characters'),
    46→  lastName: z.string().min(2, 'Last name must be at least 2 characters'),
    47→})
    48→
    49→const passwordSchema = z.object({
    50→  password: z.string()
    51→    .min(8, 'Password must be at least 8 characters')
    52→    .regex(/[A-Z]/, 'Must contain an uppercase letter')
    53→    .regex(/[a-z]/, 'Must contain a lowercase letter')
    54→    .regex(/[0-9]/, 'Must contain a number'),
    55→  confirmPassword: z.string(),
    56→}).refine(data => data.password === data.confirmPassword, {
    57→  message: 'Passwords do not match',
    58→  path: ['confirmPassword'],
    59→})
    60→
    61→type NameValues = z.infer<typeof nameSchema>
    62→type PasswordValues = z.infer<typeof passwordSchema>
    63→
    64→const STEPS = [

    66→  { id: 2, label: 'Birthday', icon: Calendar },
    67→  { id: 3, label: 'Password', icon: Lock },
    68→  { id: 4, label: 'Review', icon: Shield },
    69→]
    70→
    71→export function RegisterForm() {

    80→  const [lastName, setLastName] = useState('')

   113→  const checkTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null)

   115→    if (!fullEmail || !isValidUsername) {
   116→      setEmailStatus('idle')
   117→      return

   120→    if (checkTimerRef.current) clearTimeout(checkTimerRef.current)

   127→        setEmailStatus('idle')
   128→      }

   137→      toast.error('Please enter your first and last name (min 2 chars each)')
   138→      return

   140→    if (!fullEmail || !isValidUsername) {
   141→      toast.error('Please enter a valid email username')

   145→      toast.error('Your chosen email is taken. Please pick another one')

   147→    }

   149→    setLastName(ln)
   150→    setStep(2)
   151→  }
   152→

   155→      toast.error('Please fill in your complete date of birth')

   158→    const dob = new Date(year, month - 1, day)
   159→    const age = Math.floor((Date.now() - dob.getTime()) / (365.25 * 24 * 60 * 60 * 1000))

   167→  const handlePasswordNext = () => {

   169→      if (valid) setStep(4)
   170→    })
   171→  }
   172→

   174→    if (!agreedToTerms) {
   175→      toast.error('You must agree to the terms and conditions')
   176→      return
   177→    }
   178→    if (!fullEmail || emailStatus === 'taken') {
   179→      toast.error('Please choose an available email address')
   180→      return

   205→      setIsLoading(false)
   206→    }
   207→  }
   208→
   209→  return (
   210→    <motion.div
   211→      initial={{ opacity: 0, y: 20, scale: 0.95 }}
   212→      animate={{ opacity: 1, y: 0, scale: 1 }}
   213→      transition={{ duration: 0.5, ease: 'easeOut' }}
   214→    >
   215→      <div className="p-6 sm:p-8 flex flex-col overflow-y-auto">
   216→        {/* Personal / Business Tab Switcher — Business tab hidden on mobile */}
   217→        <div className="hidden lg:flex rounded-xl bg-gray-100 dark:bg-gray-800 p-1 mb-6">
   218→          <button
   219→            className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg text-sm font-medium text-[#1F1F1F] dark:text-white bg-white dark:bg-gray-700 shadow-sm transition-all"
   220→          >
   221→            <User className="w-3.5 h-3.5" />
   222→            Personal
   223→          </button>
   224→          <button
   225→            onClick={() => setAuthView('business-register')}
   226→            className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg text-sm font-medium text-gray-500 dark:text-gray-400 hover:text-[#1F1F1F] dark:hover:text-white transition-all"
   227→          >
   228→            <Building2 className="w-3.5 h-3.5" />
   229→            Business
   230→          </button>
   231→        </div>
   232→
   233→          {/* Progress Steps */}
   234→          <div className="flex items-center justify-between mb-8 px-2 sm:px-4">

   252→                    </div>
   253→                    <span className={`text-[10px] sm:text-xs mt-1 font-medium ${
   254→                      isActive ? 'text-[#4285F4]' : isCompleted ? 'text-[#34A853]' : 'text-gray-400'
   255→                    }`}>
   256→                      {s.label}
   257→                    </span>

   259→                  {i < STEPS.length - 1 && (

   262→                    }`} />
   263→                  )}
   264→                </div>
   265→              )
   266→            })}
   267→          </div>
   268→
   269→          {/* Step Content */}
   270→          <AnimatePresence mode="wait">
   271→            {/* STEP 1: Name + Email */}
   272→            {step === 1 && (
   273→              <motion.div
   274→                key="step1"
   275→                initial={{ opacity: 0, x: 20 }}
   276→                animate={{ opacity: 1, x: 0 }}
   277→                exit={{ opacity: 0, x: -20 }}
   278→                transition={{ duration: 0.25 }}
   279→              >
   280→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   281→                  What&apos;s your name?
   282→                </h2>
   283→                <p className="text-sm text-[#444746] dark:text-gray-400 mb-5">
   284→                  Choose your name and pick your email address
   285→                </p>
   286→                <div className="space-y-4">
   287→                  <div className="space-y-2">
   288→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">First name</label>
   289→                    <div className="relative">
   290→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   291→                      <Input
   292→                        {...nameForm.register('firstName')}
   293→                        placeholder="John"
   294→                        className="h-11 rounded-xl pl-10 border-gray-200 dark:border-gray-700 focus:border-[#4285F4] focus:ring-[#4285F4]/20"
   295→                        autoFocus
   296→                      />
   297→                    </div>
   298→                    {nameForm.formState.errors.firstName && (
   299→                      <p className="text-xs text-red-500">{nameForm.formState.errors.firstName.message}</p>
   300→                    )}
   301→                  </div>
   302→                  <div className="space-y-2">
   303→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">Last name</label>
   304→                    <div className="relative">
   305→                      <User className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />
   306→                      <Input
   307→                        {...nameForm.register('lastName')}
   308→                        placeholder="Doe"
   309→                        className="h-11 rounded-xl pl-10 border-gray-200 dark:border-gray-700 focus:border-[#4285F4] focus:ring-[#4285F4]/20"
   310→                      />
   311→                    </div>
   312→                    {nameForm.formState.errors.lastName && (
   313→                      <p className="text-xs text-red-500">{nameForm.formState.errors.lastName.message}</p>
   314→                    )}
   315→                  </div>
   316→
   317→                  {/* EMAIL CHOOSER */}
   318→                  <div className="space-y-2">
   319→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">
   320→                      Email address

   322→                    <div className="relative">
   323→                      <Mail className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-gray-400 pointer-events-none" />

   327→                        placeholder="yourname"
   328→                        className="h-11 rounded-xl pl-10 pr-20 border-gray-200 dark:border-gray-700 focus:border-[#4285F4] focus:ring-[#4285F4]/20"
   329→                      />

   331→                        @ezy.af
   332→                      </span>
   333→                    </div>

   336→                      <div className="flex items-center gap-1.5">

   341→                          <>

   354→                  </div>
   355→                </div>
   356→              </motion.div>

   359→            {/* STEP 2: Birthday */}

   361→              <motion.div
   362→                key="step2"
   363→                initial={{ opacity: 0, x: 20 }}
   364→                animate={{ opacity: 1, x: 0 }}
   365→                exit={{ opacity: 0, x: -20 }}
   366→                transition={{ duration: 0.25 }}
   367→              >
   368→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   369→                  When&apos;s your birthday?

   416→                </div>
   417→              </motion.div>
   418→            )}
   419→
   420→            {/* STEP 3: Password */}
   421→            {step === 3 && (
   422→              <motion.div
   423→                key="step3"
   424→                initial={{ opacity: 0, x: 20 }}
   425→                animate={{ opacity: 1, x: 0 }}
   426→                exit={{ opacity: 0, x: -20 }}
   427→                transition={{ duration: 0.25 }}
   428→              >
   429→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">

   495→                  </div>
   496→                </div>

   499→
   500→            {/* STEP 4: Review */}
   501→            {step === 4 && (

   554→            {step > 1 && (

   576→          {/* Back to login */}

   578→            <button onClick={() => setAuthView('login')}
