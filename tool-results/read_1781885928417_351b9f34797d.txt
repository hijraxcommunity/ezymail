    50→function isImageFile(name: string) {
    51→  return /\.(jpg|jpeg|png|gif|webp|svg|bmp|ico)$/i.test(name)
    52→}

    54→function formatFileSize(bytes?: string | number) {
    55→  if (!bytes) return ''
    56→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
    57→  if (isNaN(n)) return bytes
    58→  if (n < 1024) return `${n} B`
    59→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
    60→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
    61→}
    62→
    63→function getInitials(user: { firstName?: string; lastName?: string } | null | undefined, fallback = 'U') {
    64→  if (!user) return fallback
    65→  return `${user.firstName?.charAt(0) || ''}${user.lastName?.charAt(0) || ''}`.toUpperCase() || fallback
    66→}

   163→

   165→    <div className="space-y-3">
   166→      <div className="flex items-center gap-2">
   167→        <Paperclip className="w-4 h-4 text-gray-500" />
   168→        <span className="text-sm font-medium text-[#1F1F1F] dark:text-white">
   169→          {attachments.length} attachment{attachments.length > 1 ? 's' : ''}
   170→        </span>
   171→      </div>
   172→      {images.length > 0 && (
   173→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   174→          {images.map((att, idx) => (
   175→            <a key={idx} href={att.data || att.url} target="_blank" rel="noopener noreferrer"
   176→              className="group relative rounded-xl overflow-hidden border border-gray-200 dark:border-gray-700 bg-gray-50 dark:bg-gray-900 aspect-square flex items-center justify-center">

   179→              <div className="absolute bottom-0 inset-x-0 bg-gradient-to-t from-black/60 to-transparent p-1.5 opacity-0 group-hover:opacity-100 transition-opacity">

   187→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   188→          {docs.map((att, idx) => (
   189→            <a key={idx} href={att.data || att.url} target="_blank" rel="noopener noreferrer"

   209→function LabelManager({ emailId }: { emailId: string }) {
   210→  const { labels, emailLabelsMap, setEmailLabels, setLabels, addLabel } = useAppStore()
   211→  const [showCreate, setShowCreate] = useState(false)
   212→  const [newName, setNewName] = useState('')
   213→  const [newColor, setNewColor] = useState('#4285F4')
   214→  const [isAdding, setIsAdding] = useState<string | null>(null)
   215→  const [isRemoving, setIsRemoving] = useState<string | null>(null)
   216→
   217→  const currentLabels = emailLabelsMap[emailId] || []
   218→  const appliedIds = new Set(currentLabels.map((l) => l.id))
   219→
   220→  const handleToggleLabel = async (label: Label) => {
   221→    const isApplied = appliedIds.has(label.id)
   222→
   223→    if (isApplied) {
   224→      // Remove label from email
   225→      setIsRemoving(label.id)
   226→      try {
   227→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   228→          method: 'DELETE',
   229→          headers: { 'Content-Type': 'application/json' },
   230→          body: JSON.stringify({ labelIds: [label.id] }),
   231→        })
   232→        const data = await res.json()
   233→        if (data.success) {
   234→          setEmailLabels(emailId, data.data)
   235→        }
   236→      } catch {
   237→        toast.error('Failed to remove label')
   238→      } finally {
   239→        setIsRemoving(null)
   240→      }
   241→    } else {
   242→      // Add label to email
   243→      setIsAdding(label.id)
   244→      try {
   245→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   246→          method: 'POST',
   247→          headers: { 'Content-Type': 'application/json' },
   248→          body: JSON.stringify({ labelIds: [label.id] }),
   249→        })
   250→        const data = await res.json()
   251→        if (data.success) {
   252→          setEmailLabels(emailId, data.data)
   253→        }
   254→      } catch {
   255→        toast.error('Failed to add label')
   256→      } finally {
   257→        setIsAdding(null)
   258→      }
   259→    }
   260→  }
   261→
   262→  const handleCreateLabel = async () => {
   263→    const name = newName.trim()
   264→    if (!name) return
   265→
   266→    try {
   267→      const res = await fetch('/api/labels', {
   268→        method: 'POST',
   269→        headers: { 'Content-Type': 'application/json' },
   270→        body: JSON.stringify({ name, color: newColor }),
   271→      })
   272→      const data = await res.json()
   273→      if (data.success) {
   274→        addLabel(data.data)
   275→        setNewName('')
   276→        setNewColor('#4285F4')
   277→        setShowCreate(false)
   278→        toast.success('Label created')
   279→      } else {
   280→        toast.error(data.error || 'Failed to create label')
   281→      }
   282→    } catch {
   283→      toast.error('Failed to create label')
   284→    }
   285→  }
   286→
   287→  return (
   288→    <DropdownMenuContent align="end" className="w-56 p-0">
   289→      <DropdownMenuLabel className="px-3 py-2 text-xs font-semibold uppercase tracking-wider text-gray-400">
   290→        Labels
   291→      </DropdownMenuLabel>
   292→      <DropdownMenuSeparator />
   293→      <div className="max-h-48 overflow-y-auto">
   294→        {labels.map((label) => {
   295→          const isApplied = appliedIds.has(label.id)
   296→          const isLoading = isAdding === label.id || isRemoving === label.id
   297→          return (
   298→            <DropdownMenuItem
   299→              key={label.id}
   300→              onClick={() => handleToggleLabel(label)}
   301→              disabled={isLoading}
   302→              className="flex items-center gap-2.5 px-3 py-2 cursor-pointer"
   303→            >
   304→              {isLoading ? (
   305→                <div className="w-4 h-4 border-2 border-gray-300 border-t-transparent rounded-full animate-spin" />
   306→              ) : isApplied ? (
   307→                <div
   308→                  className="w-4 h-4 rounded-sm flex items-center justify-center shrink-0"
   309→                  style={{ backgroundColor: label.color }}
   310→                >
   311→                  <Check className="w-3 h-3 text-white" />
   312→                </div>
   313→              ) : (
   314→                <div
   315→                  className="w-4 h-4 rounded-sm border-2 shrink-0"
   316→                  style={{ borderColor: label.color }}
   317→                />
   318→              )}
   319→              <span className="flex-1 text-sm truncate">{label.name}</span>
   320→            </DropdownMenuItem>
   321→          )
   322→        })}
   323→        {labels.length === 0 && (
   324→          <div className="px-3 py-4 text-center text-sm text-gray-400">
   325→            No labels yet
   326→          </div>
   327→        )}
   328→      </div>
   329→      <DropdownMenuSeparator />
   330→      <div className="p-2">
   331→        {!showCreate ? (
   332→          <button
   333→            onClick={() => setShowCreate(true)}
   334→            className="w-full flex items-center gap-2 px-2 py-1.5 text-sm text-[#4285F4] hover:bg-[#D3E3FD]/50 dark:hover:bg-[#4285F4]/10 rounded-lg transition-colors"
   335→          >
   336→            <Plus className="w-3.5 h-3.5" />
   337→            Create new label
   338→          </button>
   339→        ) : (
   340→          <div className="space-y-2 p-1">
   341→            <Input
   342→              value={newName}
   343→              onChange={(e) => setNewName(e.target.value)}
   344→              onKeyDown={(e) => {
   345→                if (e.key === 'Enter') handleCreateLabel()
   346→                if (e.key === 'Escape') setShowCreate(false)
   347→              }}
   348→              placeholder="Label name"
   349→              className="h-7 text-xs"
   350→              autoFocus
   351→            />
   352→            <div className="flex items-center gap-1.5">
   353→              {['#4285F4', '#EA4335', '#FBBC04', '#34A853', '#FF6D01', '#E91E63'].map((c) => (
   354→                <button
   355→                  key={c}
   356→                  type="button"
   357→                  onClick={() => setNewColor(c)}
   358→                  className={`w-4 h-4 rounded-full transition-transform ${newColor === c ? 'scale-125 ring-1 ring-offset-1 ring-gray-400' : ''}`}
   359→                  style={{ backgroundColor: c }}
   360→                />
   361→              ))}
   362→            </div>
   363→            <div className="flex gap-1">
   364→              <button
   365→                onClick={handleCreateLabel}
   366→                className="flex-1 text-xs text-[#4285F4] font-medium hover:bg-[#D3E3FD]/50 rounded-md py-1 transition-colors"
   367→              >
   368→                Create
   369→              </button>
   370→              <button
   371→                onClick={() => setShowCreate(false)}
   372→                className="text-xs text-gray-500 hover:bg-gray-100 rounded-md px-2 py-1 transition-colors"
   373→              >
   374→                Cancel
   375→              </button>
   376→            </div>
   377→          </div>
   378→        )}
   379→      </div>
   380→    </DropdownMenuContent>
   381→  )
   382→}

   384→/* ═══════════════════════════════════════════
   385→   Main EmailDetail Component
   386→   ═══════════════════════════════════════════ */
   387→
   388→export function EmailDetail() {
   389→  const {
   390→    selectedEmailId,
   391→    setEmailDetailOpen,
   392→    setSelectedEmailId,
   393→    removeEmail,
   394→    updateEmail,
   395→    setReplyToEmail,
   396→    emailLabelsMap,
   397→    setEmailLabels,
   398→    currentFolder,
   399→  } = useAppStore()
   400→
   401→  const [email, setEmail] = useState<EmailWithSender | null>(null)
   402→  const [thread, setThread] = useState<EmailWithSender[]>([])
   403→  const [loading, setLoading] = useState(false)
   404→  const [expandedReplies, setExpandedReplies] = useState<Set<string>>(new Set())
   405→  const [showInfoBox, setShowInfoBox] = useState(false)
   406→
   407→  /* ─── Fetch email ─── */
   408→  const fetchEmail = useCallback(async () => {
   409→    if (!selectedEmailId) return
   410→    setLoading(true)
   411→    setShowInfoBox(false)
   412→    try {
   413→      const res = await fetch(`/api/emails/${selectedEmailId}`)
   414→      const data = await res.json()
   415→      if (res.ok) {
   416→        setEmail(data.email)
   417→        setThread(data.thread || [])
   418→        if (!data.email.isRead) {
   419→          updateEmail(data.email.id, { isRead: true })
   420→          fetch(`/api/emails/${selectedEmailId}`, {
   421→            method: 'PUT',
   422→            headers: { 'Content-Type': 'application/json' },
   423→            body: JSON.stringify({ isRead: true }),
   424→          }).catch(() => {})
   425→        }
   426→      }
   427→    } catch {
   428→      toast.error('Failed to load email')
   429→    } finally {
   430→      setLoading(false)
   431→    }
   432→  }, [selectedEmailId, updateEmail])
   433→
   434→  useEffect(() => {
   435→    if (selectedEmailId) {
   436→      fetchEmail()
   437→    } else {
   438→      setEmail(null)
   439→      setThread([])
   440→    }
   441→  }, [selectedEmailId, fetchEmail])
   442→
   443→  // Auto-expand latest message in thread when email loads
   444→  useEffect(() => {
   445→    if (thread.length > 0) {
   446→      const latestId = thread[thread.length - 1].id
   447→      setExpandedReplies(new Set([latestId]))
   448→    } else {
   449→      setExpandedReplies(new Set())
   450→    }
   451→  }, [email?.id, thread])
   452→
   453→  const toggleReply = useCallback((replyId: string) => {
   454→    setExpandedReplies((prev) => {
   455→      const next = new Set(prev)
   456→      if (next.has(replyId)) {
   457→        next.delete(replyId)
   458→      } else {
   459→        next.add(replyId)
   460→      }
   461→      return next
   462→    })
   463→  }, [])
   464→
   465→  /* ─── Fetch email labels ─── */
   466→  useEffect(() => {
   467→    if (!selectedEmailId) return
   468→    if (emailLabelsMap[selectedEmailId]) return
   469→
   470→    fetch(`/api/emails/${selectedEmailId}/labels`)
   471→      .then((res) => res.json())
   472→      .then((data) => {
   473→        if (data.success) {
   474→          setEmailLabels(selectedEmailId, data.data)
   475→        }
   476→      })
   477→      .catch(() => {})
   478→  }, [selectedEmailId, emailLabelsMap, setEmailLabels])
   479→
   480→  /* ─── Back navigation ─── */
   481→  const handleBack = useCallback(() => {
   482→    setSelectedEmailId(null)
   483→    setEmailDetailOpen(false)
   484→  }, [setSelectedEmailId, setEmailDetailOpen])
   485→
   486→  useEffect(() => {
   487→    const onKeyDown = (e: KeyboardEvent) => {
   488→      if (e.key === 'Escape' && selectedEmailId) {
   489→        e.preventDefault()
   490→        handleBack()
   491→      }
   492→    }
   493→    window.addEventListener('keydown', onKeyDown)
   494→    return () => window.removeEventListener('keydown', onKeyDown)
   495→  }, [selectedEmailId, handleBack])
   496→
   497→  /* ─── Star toggle ─── */
   498→  const handleStar = async () => {
   499→    if (!email) return
   500→    const v = !email.isStarred
   501→    updateEmail(email.id, { isStarred: v })
   502→    setEmail({ ...email, isStarred: v })
   503→    try {
   504→      await fetch(`/api/emails/${email.id}`, {
   505→        method: 'PUT',
   506→        headers: { 'Content-Type': 'application/json' },
   507→        body: JSON.stringify({ isStarred: v }),
   508→      })
   509→    } catch {
   510→      toast.error('Failed to update')
   511→    }
   512→  }
   513→
   514→  /* ─── Archive ─── */
   515→  const handleArchive = async () => {
   516→    if (!email) return
   517→    removeEmail(email.id)
   518→    handleBack()
   519→    try {
   520→      await fetch(`/api/emails/${email.id}`, {
   521→        method: 'PUT',
   522→        headers: { 'Content-Type': 'application/json' },
   523→        body: JSON.stringify({ isArchived: true }),
   524→      })
   525→      toast.success('Email archived')
   526→    } catch {
   527→      toast.error('Failed to archive')
   528→    }
   529→  }
   530→
   531→  /* ─── Unarchive ─── */
