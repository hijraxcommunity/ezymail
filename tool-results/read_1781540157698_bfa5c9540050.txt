    38→function getPasswordStrength(password: string) {
    39→  if (!password) return { score: 0, label: '', color: '' }
    40→  let score = 0
    41→  if (password.length >= 8) score++
    42→  if (/[a-z]/.test(password)) score++
    43→  if (/[A-Z]/.test(password)) score++
    44→  if (/[0-9]/.test(password)) score++
    45→  if (/[^a-zA-Z0-9]/.test(password)) score++
    46→  if (score <= 2) return { score, label: 'Weak', color: 'bg-red-500' }
    47→  if (score <= 3) return { score, label: 'Medium', color: 'bg-amber-500' }
    48→  return { score, label: 'Strong', color: 'bg-emerald-500' }
    49→}
    50→
    51→const passwordSchema = z.object({
    52→  password: z.string()
    53→    .min(8, 'Password must be at least 8 characters')
    54→    .regex(/[A-Z]/, 'Must contain an uppercase letter')
    55→    .regex(/[a-z]/, 'Must contain a lowercase letter')
    56→    .regex(/[0-9]/, 'Must contain a number'),
    57→  confirmPassword: z.string(),
    58→}).refine(data => data.password === data.confirmPassword, {
    59→  message: 'Passwords do not match',
    60→  path: ['confirmPassword'],
    61→})
    62→
    63→type PasswordValues = z.infer<typeof passwordSchema>
    64→

   217→            <User className="w-3.5 h-3.5" />
   218→            Personal

   221→            className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg text-sm font-medium text-white bg-[#34A853] shadow-sm transition-all"

   284→                    Business Name
   285→                  </label>

   287→                    value={businessName}
   288→                    onChange={(e) => setBusinessName(e.target.value)}
   289→                    placeholder="Acme Inc."
   290→                    className="h-11 rounded-xl border-gray-200 dark:border-gray-700 focus:border-[#34A853] focus:ring-[#34A853]/20"
   291→                    autoFocus
   292→                  />

   296→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">

   302→                      placeholder="John"

   304→                    />
   305→                  </div>
   306→                  <div className="space-y-2">
   307→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">
   308→                      Last Name
   309→                    </label>
   310→                    <Input
   311→                      value={lastName}
   312→                      onChange={(e) => setLastName(e.target.value)}
   313→                      placeholder="Doe"
   314→                      className="h-11 rounded-xl border-gray-200 dark:border-gray-700 focus:border-[#34A853] focus:ring-[#34A853]/20"
   315→                    />
   316→                  </div>
   317→                </div>
   318→              </div>
   319→            </motion.div>
   320→          )}
   321→
   322→          {/* STEP 2: Email */}
   323→          {step === 2 && (
   324→            <motion.div
   325→              key="step2"
   326→              initial={{ opacity: 0, x: 20 }}
   327→              animate={{ opacity: 1, x: 0 }}
   328→              exit={{ opacity: 0, x: -20 }}
   329→              transition={{ duration: 0.25 }}
   330→            >
   331→              <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   332→                Your business email
   333→              </h2>
   334→              <p className="text-sm text-[#444746] dark:text-gray-400 mb-5">
   335→                Choose your professional email address
   336→              </p>
   337→              <div className="space-y-4">
   338→                <div className="space-y-2">
   339→                  <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">
   340→                    Business Email
