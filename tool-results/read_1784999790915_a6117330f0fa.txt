    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   196→  const addToChip = async (email: string, type: 'to' | 'cc' | 'bcc') => {

   208→      setChips(prev => [...prev, { email: trimmed, name: (contact as { name: string; avatar?: string | null }).name || '', avatar: (contact as { avatar?: string | null })?.avatar || null }])

   211→        const existing = prev.split(',').map(s => s.trim()).filter(Boolean)
   212→        if (!existing.includes(trimmed)) return [...existing, trimmed].join(', ')

   710→                  const data = JSON.parse(xhr.responseText)

   713→                  reject(new Error('Invalid response from server'))
   714→                }
   715→              } else {
   716→                try {
   717→                  const err = JSON.parse(xhr.responseText)
   718→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   719→                } catch {
   720→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   721→                }
   722→              }
   723→            })

   725→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))

   727→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))

   729→          })
