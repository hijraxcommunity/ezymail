    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   175→      setChips(prev => [...prev, { email: trimmed, name: (contact as { name: string; avatar?: string | null }).name || '', avatar: (contact as { avatar?: string | null })?.avatar || null }])

   180→        return prev
   181→      })
   182→      return

   435→  const processRecipients = (value: string): string => {
   436→    return value
   437→      .split(',')
   438→      .map(r => r.trim())
   439→      .filter(Boolean)
   440→      .map(r => {

   445→        if (/^[a-zA-Z0-9._-]+$/.test(r)) return `${r}@ezy.af`
   446→        // If it contains spaces or looks like a name, return as-is (will fail validation)
   447→        return r
   448→      })
   449→      .join(', ')
   450→  }

   452→  const actuallySendEmail = useCallback(async (data: PendingSendData) => {

   460→      }

   462→      if (data.bcc.trim()) payload.bcc = processRecipients(data.bcc)

   464→      if (data.attachments && data.attachments.length > 0) {
   465→        payload.attachments = data.attachments.map(a => ({ name: a.name, url: a.url, size: a.size, type: a.type, data: a.data }))
   466→      }
   467→      if (data.priority === 'high') payload.priority = 'high'
   468→      if (data.scheduledAt) payload.scheduledAt = data.scheduledAt

   470→      const res = await fetch('/api/emails', {
   471→        method: 'POST',
   472→        headers: { 'Content-Type': 'application/json' },
   473→        body: JSON.stringify(payload),
   474→      })
   475→
   476→      if (!res.ok) {
   477→        const errData = await res.json()
   478→        toast.error(errData.error || 'Failed to send email')
   479→      }
   480→    } catch {
   481→      toast.error('Failed to send email')

   495→    if (!pending) return
   496→    pendingSendRef.current = null
   497→    setUndoCountdown(0)
   498→
   499→    // If email was already sent (has sentEmailId), delete it
   500→    if (pending.sentEmailId) {

   502→    }

   594→                  const data = JSON.parse(xhr.responseText)
   595→                  resolve(data.files)
   596→                } catch {
   597→                  reject(new Error('Invalid response from server'))
   598→                }
   599→              } else {
   600→                try {
   601→                  const err = JSON.parse(xhr.responseText)
   602→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   603→                } catch {
   604→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   605→                }
   606→              }
   607→            })
   608→            xhr.addEventListener('error', () => reject(new Error('Network error — check your connection')))
   609→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))

   611→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))
   612→            xhr.send(formData)

   618→          setUploadProgress(0)

   620→        }

   622→        setUploadProgress(0)
   623→      }

   626→        to, cc, bcc, subject, html,

   648→        const sendResult = await actuallySendEmailWithResponse(data)

   651→          // Delete draft if editing one
   652→          if (editDraftEmail) {
   653→            fetch(`/api/emails/${editDraftEmail.id}`, { method: 'DELETE' }).catch(() => {})

   674→        }, 5000)
   675→      }
   676→    } finally {
   677→      isSendingRef.current = false
   678→    }
