    61→function ToolbarButton({
    62→  active,
    63→  onClick,
    64→  disabled,
    65→  children,
    66→  title,
    67→}: {
    68→  active?: boolean
    69→  onClick: () => void
    70→  disabled?: boolean
    71→  children: React.ReactNode
    72→  title?: string
    73→}) {
    74→  const handleMouseDown = useCallback(
    75→    (e: React.MouseEvent) => {
    76→      e.preventDefault()
    77→      onClick()
    78→    },
    79→    [onClick],
    80→  )
    81→
    82→  return (
    83→    <button
    84→      type="button"
    85→      title={title}
    86→      disabled={disabled}
    87→      className={cn(
    88→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    89→        'hover:bg-accent hover:text-accent-foreground',
    90→        'active:scale-95',
    91→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    92→        active && 'bg-accent text-accent-foreground'
    93→      )}
    94→      onMouseDown={handleMouseDown}
    95→    >
    96→      {children}
    97→    </button>
    98→  )
    99→}

   383→  const handleSend = async (scheduledAtDate?: Date | null) => {

   400→            e.cc = `"${part}" is not a valid @ezy.af address`

   402→          }
   403→        }
   404→      }
   405→      // Validate BCC recipients if provided
   406→      if (showBcc && bcc.trim()) {
   407→        const bccParts = bcc.split(',').map(r => r.trim()).filter(Boolean)
   408→        for (const part of bccParts) {
   409→          const processed = processRecipients(part)
   410→          if (!processed.endsWith('@ezy.af')) {
   411→            e.bcc = `"${part}" is not a valid @ezy.af address`
   412→            break
   413→          }
   414→        }
   415→      }
   416→      setErrors(e)
   417→      if (Object.keys(e).length > 0) return
   418→
   419→      const html = editor?.getHTML() || ''
   420→      const currentAttachments = attachments
   421→
   422→      // Upload attachments FIRST
   423→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined
   424→      if (currentAttachments.length > 0) {
   425→        setIsUploading(true)
   426→        setUploadProgress(0)
   427→        try {
   428→          const result = await new Promise<Array<{ name: string; url: string; size: number; type: string; data: string }>>((resolve, reject) => {
   429→            const formData = new FormData()
   430→            currentAttachments.forEach(file => formData.append('files', file))
   431→            const xhr = new XMLHttpRequest()
   432→            xhr.open('POST', '/api/upload')
   433→            xhr.withCredentials = true
   434→            xhr.upload.addEventListener('progress', (ev) => {
   435→              if (ev.lengthComputable) {
   436→                setUploadProgress(Math.round((ev.loaded / ev.total) * 100))
   437→              }
   438→            })
   439→            xhr.addEventListener('load', () => {
   440→              if (xhr.status >= 200 && xhr.status < 300) {
   441→                try {
   442→                  const data = JSON.parse(xhr.responseText)
   443→                  resolve(data.files)
   444→                } catch {
   445→                  reject(new Error('Invalid response from server'))
   446→                }
   447→              } else {
   448→                try {
   449→                  const err = JSON.parse(xhr.responseText)
   450→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   451→                } catch {
   452→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   453→                }
   454→              }
   455→            })
   456→            xhr.addEventListener('error', () => reject(new Error('Network error — check your connection')))
   457→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))
   458→            xhr.timeout = 120000 // 2 minute timeout
   459→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))
   460→            xhr.send(formData)
   461→          })
   462→          uploadedFiles = result
   463→        } catch (err) {
   464→          toast.error(err instanceof Error ? err.message : 'Failed to upload attachments')
   465→          setIsUploading(false)
   466→          setUploadProgress(0)
   467→          return
   468→        }
   469→        setIsUploading(false)
   470→        setUploadProgress(0)
   471→      }
   472→
   473→      const data: PendingSendData = {
   474→        to, cc, bcc, subject, html,
   475→        replyToId: replyToEmail?.id,
   476→        attachments: uploadedFiles,
   477→        priority,
   478→        scheduledAt: scheduledAtDate ? scheduledAtDate.toISOString() : null,
   479→      }
   480→      pendingSendRef.current = data
   481→
   482→      // Close compose modal
   483→      setComposeOpen(false)
   484→      setShowSchedulePopover(false)
   485→
   486→      if (scheduledAtDate) {
   487→        // Schedule send — send immediately to API (no undo)
   488→        toast.success(`Email scheduled for ${format(scheduledAtDate, 'MMM d, yyyy h:mm a')}`)
   489→        actuallySendEmail(data)
   490→      } else {
   491→        // Normal send with undo
   492→        const sendResult = await actuallySendEmailWithResponse(data)
   493→        if (sendResult?.emailId) {
   494→          pendingSendRef.current = { ...data, sentEmailId: sendResult.emailId }
   495→        }
   496→
   497→        toast('Message sent.', {
   498→          description: undoCountdown > 0 ? `Undo available (${undoCountdown}s)` : undefined,
   499→          action: {
   500→            label: 'Undo',
   501→            onClick: handleUndo,
   502→          },
   503→          duration: 5000,
   504→        })
   505→
   506→        startUndoCountdown()
   507→
   508→        undoTimerRef.current = setTimeout(() => {
   509→          const payload = pendingSendRef.current
   510→          if (!payload) return
   511→          pendingSendRef.current = null
   512→          if (countdownTimerRef.current) clearInterval(countdownTimerRef.current)
   513→          setUndoCountdown(0)
   514→        }, 5000)
   515→      }
   516→    } finally {
   517→      isSendingRef.current = false
   518→    }
   519→  }

   521→  const actuallySendEmailWithResponse = useCallback(async (data: PendingSendData) => {
   522→    try {
   523→      const finalTo = processRecipients(data.to)
   524→      const payload: Record<string, unknown> = {
   525→        to: finalTo,
   526→        subject: data.subject,
   527→        body: data.html.replace(/<[^>]*>/g, ''),
   528→        bodyHtml: data.html,
   529→      }
   530→      if (data.cc.trim()) payload.cc = processRecipients(data.cc)
   531→      if (data.bcc.trim()) payload.bcc = processRecipients(data.bcc)
   532→      if (data.replyToId) payload.replyToId = data.replyToId
   533→      if (data.attachments && data.attachments.length > 0) {
   534→        payload.attachments = data.attachments.map(a => ({ name: a.name, url: a.url, size: a.size, type: a.type, data: a.data }))
   535→      }
   536→      if (data.priority === 'high') payload.priority = 'high'
   537→      payload.sentAt = new Date().toISOString()
   538→
   539→      const res = await fetch('/api/emails', {
   540→        method: 'POST',
   541→        headers: { 'Content-Type': 'application/json' },
   542→        body: JSON.stringify(payload),
   543→      })
   544→
   545→      if (!res.ok) {
   546→        const errData = await res.json()
   547→        toast.error(errData.error || 'Failed to send email')
   548→        return null
   549→      }
   550→
   551→      const responseData = await res.json()
   552→      return { emailId: responseData.email?.id }
   553→    } catch {
   554→      toast.error('Failed to send email')
   555→      return null
   556→    }
   557→  }, [])
   558→
   559→  const handleLink = useCallback(() => {
