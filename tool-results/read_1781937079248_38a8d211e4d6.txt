    61→function ToolbarButton({
    62→  active,
    63→  onClick,
    64→  disabled,
    65→  children,
    66→  title,
    67→}: {
    68→  active?: boolean
    69→  onClick: () => void
    70→  disabled?: boolean
    71→  children: React.ReactNode
    72→  title?: string
    73→}) {
    74→  const handleMouseDown = useCallback(
    75→    (e: React.MouseEvent) => {
    76→      e.preventDefault()
    77→      onClick()
    78→    },
    79→    [onClick],
    80→  )
    81→
    82→  return (
    83→    <button
    84→      type="button"
    85→      title={title}
    86→      disabled={disabled}
    87→      className={cn(
    88→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    89→        'hover:bg-accent hover:text-accent-foreground',
    90→        'active:scale-95',
    91→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    92→        active && 'bg-accent text-accent-foreground'
    93→      )}
    94→      onMouseDown={handleMouseDown}
    95→    >
    96→      {children}
    97→    </button>
    98→  )
    99→}

   404→  const handleSend = async (scheduledAtDate?: Date | null) => {

   430→          const processed = processRecipients(part)
   431→          if (!processed.endsWith('@ezy.af')) {
   432→            e.bcc = `"${part}" is not a valid @ezy.af address`
   433→            break
   434→          }
   435→        }
   436→      }
   437→      setErrors(e)
   438→      if (Object.keys(e).length > 0) return
   439→
   440→      const html = editor?.getHTML() || ''
   441→      const currentAttachments = attachments
   442→
   443→      // Upload attachments FIRST
   444→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined
   445→      if (currentAttachments.length > 0) {
   446→        setIsUploading(true)
   447→        setUploadProgress(0)
   448→        try {
   449→          const result = await new Promise<Array<{ name: string; url: string; size: number; type: string; data: string }>>((resolve, reject) => {
   450→            const formData = new FormData()
   451→            currentAttachments.forEach(file => formData.append('files', file))
   452→            const xhr = new XMLHttpRequest()
   453→            xhr.open('POST', '/api/upload')
   454→            xhr.withCredentials = true
   455→            xhr.upload.addEventListener('progress', (ev) => {
   456→              if (ev.lengthComputable) {
   457→                setUploadProgress(Math.round((ev.loaded / ev.total) * 100))
   458→              }
   459→            })
   460→            xhr.addEventListener('load', () => {
   461→              if (xhr.status >= 200 && xhr.status < 300) {
   462→                try {
   463→                  const data = JSON.parse(xhr.responseText)
   464→                  resolve(data.files)
   465→                } catch {
   466→                  reject(new Error('Invalid response from server'))
   467→                }
   468→              } else {
   469→                try {
   470→                  const err = JSON.parse(xhr.responseText)
   471→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   472→                } catch {
   473→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   474→                }
   475→              }
   476→            })
   477→            xhr.addEventListener('error', () => reject(new Error('Network error — check your connection')))
   478→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))
   479→            xhr.timeout = 120000 // 2 minute timeout
   480→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))
   481→            xhr.send(formData)
   482→          })
   483→          uploadedFiles = result
   484→        } catch (err) {
   485→          toast.error(err instanceof Error ? err.message : 'Failed to upload attachments')
   486→          setIsUploading(false)
   487→          setUploadProgress(0)
   488→          return
   489→        }
   490→        setIsUploading(false)
   491→        setUploadProgress(0)
   492→      }
   493→
   494→      const data: PendingSendData = {
   495→        to, cc, bcc, subject, html,
   496→        replyToId: replyToEmail?.id,
   497→        attachments: uploadedFiles,
   498→        priority,
   499→        scheduledAt: scheduledAtDate ? scheduledAtDate.toISOString() : null,
   500→      }
   501→      pendingSendRef.current = data
   502→
   503→      // Close compose modal
   504→      setComposeOpen(false)
   505→      setShowSchedulePopover(false)
   506→
   507→      if (scheduledAtDate) {
   508→        // Schedule send — send immediately to API (no undo)
   509→        toast.success(`Email scheduled for ${format(scheduledAtDate, 'MMM d, yyyy h:mm a')}`)
   510→        actuallySendEmail(data)
   511→        // Delete draft if editing one
   512→        if (editDraftEmail) {
   513→          fetch(`/api/emails/${editDraftEmail.id}`, { method: 'DELETE' }).catch(() => {})
   514→        }
   515→      } else {
   516→        // Normal send with undo
   517→        const sendResult = await actuallySendEmailWithResponse(data)
   518→        if (sendResult?.emailId) {
   519→          pendingSendRef.current = { ...data, sentEmailId: sendResult.emailId }
   520→          // Delete draft if editing one
   521→          if (editDraftEmail) {
   522→            fetch(`/api/emails/${editDraftEmail.id}`, { method: 'DELETE' }).catch(() => {})
   523→          }
   524→        }
   525→
   526→        toast('Message sent.', {
   527→          description: undoCountdown > 0 ? `Undo available (${undoCountdown}s)` : undefined,
   528→          action: {
   529→            label: 'Undo',
   530→            onClick: handleUndo,
   531→          },
   532→          duration: 5000,
   533→        })
   534→
   535→        startUndoCountdown()
   536→
   537→        undoTimerRef.current = setTimeout(() => {
   538→          const payload = pendingSendRef.current
   539→          if (!payload) return
   540→          pendingSendRef.current = null
   541→          if (countdownTimerRef.current) clearInterval(countdownTimerRef.current)
   542→          setUndoCountdown(0)
   543→        }, 5000)
   544→      }
   545→    } finally {
   546→      isSendingRef.current = false
   547→    }
   548→  }

   550→  const actuallySendEmailWithResponse = useCallback(async (data: PendingSendData) => {
   551→    try {
   552→      const finalTo = processRecipients(data.to)
   553→      const payload: Record<string, unknown> = {
   554→        to: finalTo,
   555→        subject: data.subject,
   556→        body: data.html.replace(/<[^>]*>/g, ''),
