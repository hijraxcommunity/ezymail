    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   474→  const handleSend = async (scheduledAtDate?: Date | null) => {

   487→        const ccParts = cc.split(',').map(r => r.trim()).filter(Boolean)
   488→        for (const part of ccParts) {
   489→          const processed = processRecipients(part)
   490→          if (!processed.endsWith('@ezy.af')) {
   491→            e.cc = `"${part}" is not a valid @ezy.af address`
   492→            break
   493→          }
   494→        }
   495→      }
   496→      // Validate BCC recipients if provided
   497→      if (showBcc && bcc.trim()) {
   498→        const bccParts = bcc.split(',').map(r => r.trim()).filter(Boolean)
   499→        for (const part of bccParts) {
   500→          const processed = processRecipients(part)
   501→          if (!processed.endsWith('@ezy.af')) {
   502→            e.bcc = `"${part}" is not a valid @ezy.af address`
   503→            break
   504→          }
   505→        }
   506→      }
   507→      setErrors(e)
   508→      if (Object.keys(e).length > 0) {
   509→        isSendingRef.current = false
   510→        return
   511→      }
   512→
   513→      const html = editor?.getHTML() || ''
   514→      const currentAttachments = attachments
   515→
   516→      // Upload attachments FIRST
   517→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined
   518→      if (currentAttachments.length > 0) {
   519→        setIsUploading(true)
   520→        setUploadProgress(0)
   521→        try {
   522→          const result = await new Promise<Array<{ name: string; url: string; size: number; type: string; data: string }>>((resolve, reject) => {
   523→            const formData = new FormData()
   524→            currentAttachments.forEach(file => formData.append('files', file))
   525→            const xhr = new XMLHttpRequest()
   526→            xhr.open('POST', '/api/upload')
   527→            xhr.withCredentials = true
   528→            xhr.upload.addEventListener('progress', (ev) => {
   529→              if (ev.lengthComputable) {
   530→                setUploadProgress(Math.round((ev.loaded / ev.total) * 100))
   531→              }
   532→            })
   533→            xhr.addEventListener('load', () => {
   534→              if (xhr.status >= 200 && xhr.status < 300) {
   535→                try {
   536→                  const data = JSON.parse(xhr.responseText)
   537→                  resolve(data.files)
   538→                } catch {
   539→                  reject(new Error('Invalid response from server'))
   540→                }
   541→              } else {
   542→                try {
   543→                  const err = JSON.parse(xhr.responseText)
   544→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   545→                } catch {
