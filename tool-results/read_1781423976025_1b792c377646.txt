    31→    31→function getPasswordStrength(password: string) {
    32→    32→  if (!password) return { score: 0, label: '', color: '' }
    33→    33→  let score = 0
    34→    34→  if (password.length >= 8) score++
    35→    35→  if (/[a-z]/.test(password)) score++
    36→    36→  if (/[A-Z]/.test(password)) score++
    37→    37→  if (/[0-9]/.test(password)) score++
    38→    38→  if (/[^a-zA-Z0-9]/.test(password)) score++
    39→    39→  if (score <= 2) return { score, label: 'Weak', color: 'bg-red-500' }
    40→    40→  if (score <= 3) return { score, label: 'Medium', color: 'bg-amber-500' }
    41→    41→  return { score, label: 'Strong', color: 'bg-emerald-500' }
    42→    42→}
