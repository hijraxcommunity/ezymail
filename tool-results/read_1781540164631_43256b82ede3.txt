    15→const MONTHS = [
    16→  'January', 'February', 'March', 'April', 'May', 'June',
    17→  'July', 'August', 'September', 'October', 'November', 'December'
    18→]
    19→
    20→function generateDays(month: number, year: number) {
    21→  if (!month || !year) return Array.from({ length: 31 }, (_, i) => i + 1)
    22→  const daysInMonth = new Date(year, month, 0).getDate()
    23→  return Array.from({ length: daysInMonth }, (_, i) => i + 1)
    24→}
    25→
    26→function generateYears() {
    27→  const currentYear = new Date().getFullYear()
    28→  return Array.from({ length: 100 }, (_, i) => currentYear - 13 - i)
    29→}
    30→
    31→function getPasswordStrength(password: string) {
    32→  if (!password) return { score: 0, label: '', color: '' }
    33→  let score = 0
    34→  if (password.length >= 8) score++
    35→  if (/[a-z]/.test(password)) score++
    36→  if (/[A-Z]/.test(password)) score++
    37→  if (/[0-9]/.test(password)) score++
    38→  if (/[^a-zA-Z0-9]/.test(password)) score++
    39→  if (score <= 2) return { score, label: 'Weak', color: 'bg-red-500' }
    40→  if (score <= 3) return { score, label: 'Medium', color: 'bg-amber-500' }
    41→  return { score, label: 'Strong', color: 'bg-emerald-500' }
    42→}
    43→
    44→const nameSchema = z.object({
    45→  firstName: z.string().min(2, 'First name must be at least 2 characters'),
    46→  lastName: z.string().min(2, 'Last name must be at least 2 characters'),
    47→})
    48→
    49→const passwordSchema = z.object({
    50→  password: z.string()
    51→    .min(8, 'Password must be at least 8 characters')
    52→    .regex(/[A-Z]/, 'Must contain an uppercase letter')
    53→    .regex(/[a-z]/, 'Must contain a lowercase letter')
    54→    .regex(/[0-9]/, 'Must contain a number'),
    55→  confirmPassword: z.string(),
    56→}).refine(data => data.password === data.confirmPassword, {
    57→  message: 'Passwords do not match',
    58→  path: ['confirmPassword'],
    59→})
    60→
    61→type NameValues = z.infer<typeof nameSchema>
    62→type PasswordValues = z.infer<typeof passwordSchema>
    63→

    67→  { id: 3, label: 'Password', icon: Lock },
    68→  { id: 4, label: 'Review', icon: Shield },
    69→]
    70→

   127→        setEmailStatus('idle')

   145→      toast.error('Your chosen email is taken. Please pick another one')

   149→    setLastName(ln)
   150→    setStep(2)

   155→      toast.error('Please fill in your complete date of birth')

   159→    const age = Math.floor((Date.now() - dob.getTime()) / (365.25 * 24 * 60 * 60 * 1000))

   167→  const handlePasswordNext = () => {

   169→      if (valid) setStep(4)
   170→    })

   175→      toast.error('You must agree to the terms and conditions')
   176→      return
   177→    }

   180→      return

   205→      setIsLoading(false)
   206→    }
   207→  }
   208→
   209→  return (
   210→    <motion.div
   211→      initial={{ opacity: 0, y: 20, scale: 0.95 }}
   212→      animate={{ opacity: 1, y: 0, scale: 1 }}
   213→      transition={{ duration: 0.5, ease: 'easeOut' }}
   214→    >
   215→      <div className="bg-white dark:bg-gray-900 sm:rounded-2xl sm:shadow-xl sm:shadow-black/5 p-6 sm:p-8 flex-1 flex flex-col justify-center sm:justify-start overflow-y-auto">

   220→          >
   221→            <User className="w-3.5 h-3.5" />
   222→            Personal
   223→          </button>
   224→          <button
   225→            onClick={() => setAuthView('business-register')}
   226→            className="flex-1 flex items-center justify-center gap-1.5 py-2 rounded-lg text-sm font-medium text-gray-500 dark:text-gray-400 hover:text-[#1F1F1F] dark:hover:text-white transition-all"
   227→          >
   228→            <Building2 className="w-3.5 h-3.5" />
   229→            Business
   230→          </button>

   252→                    </div>

   254→                      isActive ? 'text-[#4285F4]' : isCompleted ? 'text-[#34A853]' : 'text-gray-400'

   257→                    </span>

   259→                  {i < STEPS.length - 1 && (
   260→                    <div className={`w-6 sm:w-12 h-0.5 mx-1 sm:mx-2 transition-all duration-300 ${

   262→                    }`} />
   263→                  )}
   264→                </div>
   265→              )
   266→            })}
   267→          </div>

   269→          {/* Step Content */}
   270→          <AnimatePresence mode="wait">
   271→            {/* STEP 1: Name + Email */}

   273→              <motion.div
   274→                key="step1"
   275→                initial={{ opacity: 0, x: 20 }}
   276→                animate={{ opacity: 1, x: 0 }}
   277→                exit={{ opacity: 0, x: -20 }}
   278→                transition={{ duration: 0.25 }}
   279→              >
   280→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   281→                  What&apos;s your name?
   282→                </h2>

   284→                  Choose your name and pick your email address
   285→                </p>
   286→                <div className="space-y-4">
   287→                  <div className="space-y-2">
   288→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">First name</label>
   289→                    <Input
   290→                      {...nameForm.register('firstName')}
   291→                      placeholder="John"
   292→                      className="h-11 rounded-xl border-gray-200 dark:border-gray-700 focus:border-[#4285F4] focus:ring-[#4285F4]/20"
   293→                      autoFocus
   294→                    />
   295→                    {nameForm.formState.errors.firstName && (
   296→                      <p className="text-xs text-red-500">{nameForm.formState.errors.firstName.message}</p>
   297→                    )}
   298→                  </div>
   299→                  <div className="space-y-2">
   300→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">Last name</label>
   301→                    <Input
   302→                      {...nameForm.register('lastName')}
   303→                      placeholder="Doe"
   304→                      className="h-11 rounded-xl border-gray-200 dark:border-gray-700 focus:border-[#4285F4] focus:ring-[#4285F4]/20"
   305→                    />
   306→                    {nameForm.formState.errors.lastName && (
   307→                      <p className="text-xs text-red-500">{nameForm.formState.errors.lastName.message}</p>
   308→                    )}
   309→                  </div>
   310→
   311→                  {/* EMAIL CHOOSER */}
   312→                  <div className="space-y-2">
   313→                    <label className="text-sm font-medium text-[#1F1F1F] dark:text-gray-300">
   314→                      <Mail className="w-3.5 h-3.5 inline mr-1" />
   315→                      Email address
   316→                    </label>
   317→                    <div className="relative">
   318→                      <Input
   319→                        value={emailUsername}
   320→                        onChange={(e) => { setEmailUsername(e.target.value.replace(/[^a-zA-Z0-9._-]/g, '')); setEmailStatus('idle') }}
   321→                        placeholder="yourname"

   325→                        @ezy.af
   326→                      </span>

   328→                    {/* Status indicator */}

   330→                      <div className="flex items-center gap-1.5">
   331→                        {emailStatus === 'checking' && (
   332→                          <Loader2 className="w-3.5 h-3.5 animate-spin text-gray-400" />
   333→                        )}
   334→                        {emailStatus === 'available' && (
   335→                          <>

   350→              </motion.div>

   359→                exit={{ opacity: 0, x: -20 }}
   360→                transition={{ duration: 0.25 }}
   361→              >
   362→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   363→                  When&apos;s your birthday?
   364→                </h2>

   366→                  You must be 13 or older to use EzyMail

   368→                <div className="space-y-4">

   423→                <h2 className="text-lg font-semibold text-[#1F1F1F] dark:text-white mb-1">
   424→                  Create a password
   425→                </h2>
   426→                <p className="text-sm text-[#444746] dark:text-gray-400 mb-5">
   427→                  Use 8+ characters with uppercase, lowercase & numbers
