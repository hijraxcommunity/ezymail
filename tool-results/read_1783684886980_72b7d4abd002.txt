    89→function getPasswordStrength(password: string) {
    90→  if (!password) return { score: 0, label: '', color: '' }
    91→  let score = 0
    92→  if (password.length >= 8) score++
    93→  if (/[a-z]/.test(password)) score++
    94→  if (/[A-Z]/.test(password)) score++
    95→  if (/[0-9]/.test(password)) score++
    96→  if (/[^a-zA-Z0-9]/.test(password)) score++
    97→  if (score <= 2) return { score, label: 'Weak', color: 'bg-[#EA4335]' }
    98→  if (score <= 3) return { score, label: 'Medium', color: 'bg-[#FBBC05]' }
    99→  return { score, label: 'Strong', color: 'bg-[#34A853]' }
   100→}

   102→function formatDate(dateStr: string | null) {
   103→  if (!dateStr) return 'Never'
   104→  const d = new Date(dateStr)
   105→  return d.toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric', hour: '2-digit', minute: '2-digit' })
   106→}
   107→
   108→function timeAgo(dateStr: string) {
   109→  const now = Date.now()
   110→  const then = new Date(dateStr).getTime()
   111→  const diff = now - then
   112→  const mins = Math.floor(diff / 60000)
   113→  if (mins < 1) return 'Just now'
   114→  if (mins < 60) return `${mins}m ago`
   115→  const hrs = Math.floor(mins / 60)
   116→  if (hrs < 24) return `${hrs}h ago`
   117→  const days = Math.floor(hrs / 24)
   118→  return `${days}d ago`
   119→}
