    47→function isImageFile(name: string) {
    48→  return /\.(jpg|jpeg|png|gif|webp|svg|bmp|ico)$/i.test(name)
    49→}

    51→function formatFileSize(bytes?: string | number) {
    52→  if (!bytes) return ''
    53→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
    54→  if (isNaN(n)) return bytes
    55→  if (n < 1024) return `${n} B`
    56→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
    57→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
    58→}
    59→
    60→function getInitials(user: { firstName?: string; lastName?: string } | null | undefined, fallback = 'U') {
    61→  if (!user) return fallback
    62→  return `${user.firstName?.charAt(0) || ''}${user.lastName?.charAt(0) || ''}`.toUpperCase() || fallback
    63→}

   160→
   161→  return (
   162→    <div className="space-y-3">
   163→      <div className="flex items-center gap-2">
   164→        <Paperclip className="w-4 h-4 text-gray-500" />
   165→        <span className="text-sm font-medium text-[#1F1F1F] dark:text-white">
   166→          {attachments.length} attachment{attachments.length > 1 ? 's' : ''}
   167→        </span>
   168→      </div>
   169→      {images.length > 0 && (
   170→        <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   171→          {images.map((att, idx) => (
   172→            <a key={idx} href={att.data || att.url} target="_blank" rel="noopener noreferrer"
   173→              className="group relative rounded-xl overflow-hidden border border-gray-200 dark:border-gray-700 bg-gray-50 dark:bg-gray-900 aspect-square flex items-center justify-center">
   174→              <img src={att.data || att.url} alt={att.name} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-200" />
   175→              <div className="absolute inset-0 bg-black/0 group-hover:bg-black/10 transition-colors" />
   176→              <div className="absolute bottom-0 inset-x-0 bg-gradient-to-t from-black/60 to-transparent p-1.5 opacity-0 group-hover:opacity-100 transition-opacity">
   177→                <p className="text-[10px] text-white truncate">{att.name}</p>
   178→              </div>
   179→            </a>

   181→        </div>

   184→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2">
   185→          {docs.map((att, idx) => (
   186→            <a key={idx} href={att.data || att.url} target="_blank" rel="noopener noreferrer"
   187→              className="flex items-center gap-3 p-3 rounded-xl border border-gray-200 dark:border-gray-700 bg-gray-50 dark:bg-gray-900 hover:border-[#4285F4]/30 hover:bg-[#D3E3FD]/30 dark:hover:bg-[#4285F4]/5 transition-colors group">

   196→            </a>

   206→function LabelManager({ emailId }: { emailId: string }) {
   207→  const { labels, emailLabelsMap, setEmailLabels, setLabels, addLabel } = useAppStore()
   208→  const [showCreate, setShowCreate] = useState(false)
   209→  const [newName, setNewName] = useState('')
   210→  const [newColor, setNewColor] = useState('#4285F4')
   211→  const [isAdding, setIsAdding] = useState<string | null>(null)
   212→  const [isRemoving, setIsRemoving] = useState<string | null>(null)
   213→
   214→  const currentLabels = emailLabelsMap[emailId] || []
   215→  const appliedIds = new Set(currentLabels.map((l) => l.id))
   216→
   217→  const handleToggleLabel = async (label: Label) => {
   218→    const isApplied = appliedIds.has(label.id)
   219→
   220→    if (isApplied) {
   221→      // Remove label from email
   222→      setIsRemoving(label.id)
   223→      try {
   224→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   225→          method: 'DELETE',
   226→          headers: { 'Content-Type': 'application/json' },
   227→          body: JSON.stringify({ labelIds: [label.id] }),
   228→        })
   229→        const data = await res.json()
   230→        if (data.success) {
   231→          setEmailLabels(emailId, data.data)
   232→        }
   233→      } catch {
   234→        toast.error('Failed to remove label')
   235→      } finally {
   236→        setIsRemoving(null)
   237→      }
   238→    } else {
   239→      // Add label to email
   240→      setIsAdding(label.id)
   241→      try {
   242→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   243→          method: 'POST',
   244→          headers: { 'Content-Type': 'application/json' },
   245→          body: JSON.stringify({ labelIds: [label.id] }),
   246→        })
   247→        const data = await res.json()
   248→        if (data.success) {
   249→          setEmailLabels(emailId, data.data)
   250→        }
   251→      } catch {
   252→        toast.error('Failed to add label')
   253→      } finally {
   254→        setIsAdding(null)
   255→      }
   256→    }
   257→  }
   258→
   259→  const handleCreateLabel = async () => {
   260→    const name = newName.trim()
   261→    if (!name) return
   262→
   263→    try {
   264→      const res = await fetch('/api/labels', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify({ name, color: newColor }),
   268→      })
   269→      const data = await res.json()
   270→      if (data.success) {
   271→        addLabel(data.data)
   272→        setNewName('')
   273→        setNewColor('#4285F4')
   274→        setShowCreate(false)
   275→        toast.success('Label created')
   276→      } else {
   277→        toast.error(data.error || 'Failed to create label')
   278→      }
   279→    } catch {
   280→      toast.error('Failed to create label')
   281→    }
   282→  }
   283→
   284→  return (
   285→    <DropdownMenuContent align="end" className="w-56 p-0">
   286→      <DropdownMenuLabel className="px-3 py-2 text-xs font-semibold uppercase tracking-wider text-gray-400">
   287→        Labels
   288→      </DropdownMenuLabel>
   289→      <DropdownMenuSeparator />
   290→      <div className="max-h-48 overflow-y-auto">
   291→        {labels.map((label) => {
   292→          const isApplied = appliedIds.has(label.id)
   293→          const isLoading = isAdding === label.id || isRemoving === label.id
   294→          return (
   295→            <DropdownMenuItem
   296→              key={label.id}
   297→              onClick={() => handleToggleLabel(label)}
   298→              disabled={isLoading}
   299→              className="flex items-center gap-2.5 px-3 py-2 cursor-pointer"
   300→            >
   301→              {isLoading ? (
   302→                <div className="w-4 h-4 border-2 border-gray-300 border-t-transparent rounded-full animate-spin" />
   303→              ) : isApplied ? (
   304→                <div
   305→                  className="w-4 h-4 rounded-sm flex items-center justify-center shrink-0"
   306→                  style={{ backgroundColor: label.color }}
   307→                >
   308→                  <Check className="w-3 h-3 text-white" />
   309→                </div>
   310→              ) : (
   311→                <div
   312→                  className="w-4 h-4 rounded-sm border-2 shrink-0"
   313→                  style={{ borderColor: label.color }}
   314→                />
   315→              )}
   316→              <span className="flex-1 text-sm truncate">{label.name}</span>
   317→            </DropdownMenuItem>
   318→          )
   319→        })}
   320→        {labels.length === 0 && (
   321→          <div className="px-3 py-4 text-center text-sm text-gray-400">
   322→            No labels yet
   323→          </div>
   324→        )}
   325→      </div>
   326→      <DropdownMenuSeparator />
   327→      <div className="p-2">
   328→        {!showCreate ? (
   329→          <button
   330→            onClick={() => setShowCreate(true)}
   331→            className="w-full flex items-center gap-2 px-2 py-1.5 text-sm text-[#4285F4] hover:bg-[#D3E3FD]/50 dark:hover:bg-[#4285F4]/10 rounded-lg transition-colors"
   332→          >
   333→            <Plus className="w-3.5 h-3.5" />
   334→            Create new label
   335→          </button>
   336→        ) : (
   337→          <div className="space-y-2 p-1">
   338→            <Input
   339→              value={newName}
   340→              onChange={(e) => setNewName(e.target.value)}
   341→              onKeyDown={(e) => {
   342→                if (e.key === 'Enter') handleCreateLabel()
   343→                if (e.key === 'Escape') setShowCreate(false)
   344→              }}
   345→              placeholder="Label name"
   346→              className="h-7 text-xs"
   347→              autoFocus
   348→            />
   349→            <div className="flex items-center gap-1.5">
   350→              {['#4285F4', '#EA4335', '#FBBC04', '#34A853', '#FF6D01', '#E91E63'].map((c) => (
   351→                <button
   352→                  key={c}
   353→                  type="button"
   354→                  onClick={() => setNewColor(c)}
   355→                  className={`w-4 h-4 rounded-full transition-transform ${newColor === c ? 'scale-125 ring-1 ring-offset-1 ring-gray-400' : ''}`}
   356→                  style={{ backgroundColor: c }}
   357→                />
   358→              ))}
   359→            </div>
   360→            <div className="flex gap-1">
   361→              <button
   362→                onClick={handleCreateLabel}
   363→                className="flex-1 text-xs text-[#4285F4] font-medium hover:bg-[#D3E3FD]/50 rounded-md py-1 transition-colors"
   364→              >
   365→                Create
   366→              </button>
   367→              <button
   368→                onClick={() => setShowCreate(false)}
   369→                className="text-xs text-gray-500 hover:bg-gray-100 rounded-md px-2 py-1 transition-colors"
   370→              >
   371→                Cancel
   372→              </button>
   373→            </div>
   374→          </div>
   375→        )}
   376→      </div>
   377→    </DropdownMenuContent>
   378→  )
   379→}

   381→/* ═══════════════════════════════════════════
   382→   Main EmailDetail Component
   383→   ═══════════════════════════════════════════ */
   384→
   385→export function EmailDetail() {
   386→  const {
   387→    selectedEmailId,
   388→    setEmailDetailOpen,
   389→    setSelectedEmailId,
   390→    removeEmail,
   391→    updateEmail,
   392→    setReplyToEmail,
   393→    emailLabelsMap,
   394→    setEmailLabels,
   395→    currentFolder,
   396→  } = useAppStore()
   397→
   398→  const [email, setEmail] = useState<EmailWithSender | null>(null)
   399→  const [thread, setThread] = useState<EmailWithSender[]>([])
   400→  const [loading, setLoading] = useState(false)
   401→  const [expandedReplies, setExpandedReplies] = useState<Set<string>>(new Set())
   402→  const [showInfoBox, setShowInfoBox] = useState(false)
   403→
   404→  /* ─── Fetch email ─── */
   405→  const fetchEmail = useCallback(async () => {
   406→    if (!selectedEmailId) return
   407→    setLoading(true)
   408→    setShowInfoBox(false)
   409→    try {
   410→      const res = await fetch(`/api/emails/${selectedEmailId}`)
   411→      const data = await res.json()
   412→      if (res.ok) {
   413→        setEmail(data.email)
   414→        setThread(data.thread || [])
   415→        if (!data.email.isRead) {
   416→          updateEmail(data.email.id, { isRead: true })
   417→          fetch(`/api/emails/${selectedEmailId}`, {
   418→            method: 'PUT',
   419→            headers: { 'Content-Type': 'application/json' },
   420→            body: JSON.stringify({ isRead: true }),
   421→          }).catch(() => {})
   422→        }
   423→      }
   424→    } catch {
   425→      toast.error('Failed to load email')
   426→    } finally {
   427→      setLoading(false)
   428→    }
   429→  }, [selectedEmailId, updateEmail])
   430→
   431→  useEffect(() => {
   432→    if (selectedEmailId) {
   433→      fetchEmail()
   434→    } else {
   435→      setEmail(null)
   436→      setThread([])
   437→    }
   438→  }, [selectedEmailId, fetchEmail])
   439→
   440→  // Auto-expand latest message in thread when email loads
   441→  useEffect(() => {
   442→    if (thread.length > 0) {
   443→      const latestId = thread[thread.length - 1].id
   444→      setExpandedReplies(new Set([latestId]))
   445→    } else {
   446→      setExpandedReplies(new Set())
   447→    }
   448→  }, [email?.id, thread])
   449→
   450→  const toggleReply = useCallback((replyId: string) => {
   451→    setExpandedReplies((prev) => {
   452→      const next = new Set(prev)
   453→      if (next.has(replyId)) {
   454→        next.delete(replyId)
   455→      } else {
   456→        next.add(replyId)
   457→      }
   458→      return next
   459→    })
   460→  }, [])
   461→
   462→  /* ─── Fetch email labels ─── */
   463→  useEffect(() => {
   464→    if (!selectedEmailId) return
   465→    if (emailLabelsMap[selectedEmailId]) return
   466→
   467→    fetch(`/api/emails/${selectedEmailId}/labels`)
   468→      .then((res) => res.json())
   469→      .then((data) => {
   470→        if (data.success) {
   471→          setEmailLabels(selectedEmailId, data.data)
   472→        }
   473→      })
   474→      .catch(() => {})
   475→  }, [selectedEmailId, emailLabelsMap, setEmailLabels])
   476→
   477→  /* ─── Back navigation ─── */
   478→  const handleBack = useCallback(() => {
   479→    setSelectedEmailId(null)
   480→    setEmailDetailOpen(false)
   481→  }, [setSelectedEmailId, setEmailDetailOpen])
   482→
   483→  useEffect(() => {
   484→    const onKeyDown = (e: KeyboardEvent) => {
   485→      if (e.key === 'Escape' && selectedEmailId) {
   486→        e.preventDefault()
   487→        handleBack()
   488→      }
   489→    }
   490→    window.addEventListener('keydown', onKeyDown)
   491→    return () => window.removeEventListener('keydown', onKeyDown)
   492→  }, [selectedEmailId, handleBack])
   493→
   494→  /* ─── Star toggle ─── */
   495→  const handleStar = async () => {
   496→    if (!email) return
   497→    const v = !email.isStarred
   498→    updateEmail(email.id, { isStarred: v })
   499→    setEmail({ ...email, isStarred: v })
   500→    try {
   501→      await fetch(`/api/emails/${email.id}`, {
   502→        method: 'PUT',
   503→        headers: { 'Content-Type': 'application/json' },
   504→        body: JSON.stringify({ isStarred: v }),
   505→      })
   506→    } catch {
   507→      toast.error('Failed to update')
   508→    }
   509→  }
   510→
   511→  /* ─── Archive ─── */
   512→  const handleArchive = async () => {
   513→    if (!email) return
   514→    removeEmail(email.id)
   515→    handleBack()
   516→    try {
   517→      await fetch(`/api/emails/${email.id}`, {
   518→        method: 'PUT',
   519→        headers: { 'Content-Type': 'application/json' },
   520→        body: JSON.stringify({ isArchived: true }),
   521→      })
   522→      toast.success('Email archived')
   523→    } catch {
   524→      toast.error('Failed to archive')
   525→    }
   526→  }
   527→
   528→  /* ─── Delete ─── */
   529→  const handleDelete = async () => {
   530→    if (!email) return
   531→    if (currentFolder === 'trash') {
   532→      setShowDeleteConfirm(true)
