    54→    54→function formatFileSize(bytes?: string | number) {
    55→    55→  if (!bytes) return ''
    56→    56→  const n = typeof bytes === 'string' ? parseInt(bytes, 10) : bytes
    57→    57→  if (isNaN(n)) return bytes
    58→    58→  if (n < 1024) return `${n} B`
    59→    59→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
    60→    60→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
    61→    61→}

    63→    63→function getInitials(user: { firstName?: string; lastName?: string } | null | undefined, fallback = 'U') {

    65→    65→  return `${user.firstName?.charAt(0) || ''}${user.lastName?.charAt(0) || ''}`.toUpperCase() || fallback
    66→    66→}

   208→   208→
   209→   209→function LabelManager({ emailId }: { emailId: string }) {
   210→   210→  const { labels, emailLabelsMap, setEmailLabels, setLabels, addLabel } = useAppStore()
   211→   211→  const [showCreate, setShowCreate] = useState(false)
   212→   212→  const [newName, setNewName] = useState('')
   213→   213→  const [newColor, setNewColor] = useState('#4285F4')
   214→   214→  const [isAdding, setIsAdding] = useState<string | null>(null)
   215→   215→  const [isRemoving, setIsRemoving] = useState<string | null>(null)
   216→   216→
   217→   217→  const currentLabels = emailLabelsMap[emailId] || []
   218→   218→  const appliedIds = new Set(currentLabels.map((l) => l.id))
   219→   219→
   220→   220→  const handleToggleLabel = async (label: Label) => {
   221→   221→    const isApplied = appliedIds.has(label.id)
   222→   222→
   223→   223→    if (isApplied) {
   224→   224→      // Remove label from email
   225→   225→      setIsRemoving(label.id)
   226→   226→      try {
   227→   227→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   228→   228→          method: 'DELETE',
   229→   229→          headers: { 'Content-Type': 'application/json' },
   230→   230→          body: JSON.stringify({ labelIds: [label.id] }),
   231→   231→        })
   232→   232→        const data = await res.json()
   233→   233→        if (data.success) {
   234→   234→          setEmailLabels(emailId, data.data)
   235→   235→        }
   236→   236→      } catch {
   237→   237→        toast.error('Failed to remove label')
   238→   238→      } finally {
   239→   239→        setIsRemoving(null)
   240→   240→      }
   241→   241→    } else {
   242→   242→      // Add label to email
   243→   243→      setIsAdding(label.id)
   244→   244→      try {
   245→   245→        const res = await fetch(`/api/emails/${emailId}/labels`, {
   246→   246→          method: 'POST',
   247→   247→          headers: { 'Content-Type': 'application/json' },
   248→   248→          body: JSON.stringify({ labelIds: [label.id] }),
   249→   249→        })
   250→   250→        const data = await res.json()
   251→   251→        if (data.success) {
   252→   252→          setEmailLabels(emailId, data.data)
   253→   253→        }
   254→   254→      } catch {
   255→   255→        toast.error('Failed to add label')
   256→   256→      } finally {
   257→   257→        setIsAdding(null)
   258→   258→      }
   259→   259→    }
   260→   260→  }
   261→   261→
   262→   262→  const handleCreateLabel = async () => {
   263→   263→    const name = newName.trim()
   264→   264→    if (!name) return
   265→   265→
   266→   266→    try {
   267→   267→      const res = await fetch('/api/labels', {
   268→   268→        method: 'POST',
   269→   269→        headers: { 'Content-Type': 'application/json' },
   270→   270→        body: JSON.stringify({ name, color: newColor }),
   271→   271→      })
   272→   272→      const data = await res.json()
   273→   273→      if (data.success) {
   274→   274→        addLabel(data.data)
   275→   275→        setNewName('')
   276→   276→        setNewColor('#4285F4')
   277→   277→        setShowCreate(false)
   278→   278→        toast.success('Label created')
   279→   279→      } else {
   280→   280→        toast.error(data.error || 'Failed to create label')
   281→   281→      }
   282→   282→    } catch {
   283→   283→      toast.error('Failed to create label')
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  return (
   288→   288→    <DropdownMenuContent align="end" className="w-56 p-0">
   289→   289→      <DropdownMenuLabel className="px-3 py-2 text-xs font-semibold uppercase tracking-wider text-gray-400">
   290→   290→        Labels
   291→   291→      </DropdownMenuLabel>
   292→   292→      <DropdownMenuSeparator />
   293→   293→      <div className="max-h-48 overflow-y-auto">
   294→   294→        {labels.map((label) => {
   295→   295→          const isApplied = appliedIds.has(label.id)
   296→   296→          const isLoading = isAdding === label.id || isRemoving === label.id
   297→   297→          return (
   298→   298→            <DropdownMenuItem
   299→   299→              key={label.id}
   300→   300→              onClick={() => handleToggleLabel(label)}
   301→   301→              disabled={isLoading}
   302→   302→              className="flex items-center gap-2.5 px-3 py-2 cursor-pointer"
   303→   303→            >
   304→   304→              {isLoading ? (
   305→   305→                <div className="w-4 h-4 border-2 border-gray-300 border-t-transparent rounded-full animate-spin" />
   306→   306→              ) : isApplied ? (
   307→   307→                <div
   308→   308→                  className="w-4 h-4 rounded-sm flex items-center justify-center shrink-0"
   309→   309→                  style={{ backgroundColor: label.color }}
   310→   310→                >
   311→   311→                  <Check className="w-3 h-3 text-white" />
   312→   312→                </div>
   313→   313→              ) : (
   314→   314→                <div
   315→   315→                  className="w-4 h-4 rounded-sm border-2 shrink-0"
   316→   316→                  style={{ borderColor: label.color }}
   317→   317→                />
   318→   318→              )}
   319→   319→              <span className="flex-1 text-sm truncate">{label.name}</span>
   320→   320→            </DropdownMenuItem>
   321→   321→          )
   322→   322→        })}
   323→   323→        {labels.length === 0 && (
   324→   324→          <div className="px-3 py-4 text-center text-sm text-gray-400">
   325→   325→            No labels yet
   326→   326→          </div>
   327→   327→        )}
   328→   328→      </div>
   329→   329→      <DropdownMenuSeparator />
   330→   330→      <div className="p-2">
   331→   331→        {!showCreate ? (
   332→   332→          <button
   333→   333→            onClick={() => setShowCreate(true)}
   334→   334→            className="w-full flex items-center gap-2 px-2 py-1.5 text-sm text-[#4285F4] hover:bg-[#D3E3FD]/50 dark:hover:bg-[#4285F4]/10 rounded-lg transition-colors"
   335→   335→          >
   336→   336→            <Plus className="w-3.5 h-3.5" />
   337→   337→            Create new label
   338→   338→          </button>
   339→   339→        ) : (
   340→   340→          <div className="space-y-2 p-1">
   341→   341→            <Input
   342→   342→              value={newName}
   343→   343→              onChange={(e) => setNewName(e.target.value)}
   344→   344→              onKeyDown={(e) => {
   345→   345→                if (e.key === 'Enter') handleCreateLabel()
   346→   346→                if (e.key === 'Escape') setShowCreate(false)
   347→   347→              }}
   348→   348→              placeholder="Label name"
   349→   349→              className="h-7 text-xs"
   350→   350→              autoFocus
   351→   351→            />
   352→   352→            <div className="flex items-center gap-1.5">
