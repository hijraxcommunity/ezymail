    61→function ToolbarButton({
    62→  active,
    63→  onClick,
    64→  disabled,
    65→  children,
    66→  title,
    67→}: {
    68→  active?: boolean
    69→  onClick: () => void
    70→  disabled?: boolean
    71→  children: React.ReactNode
    72→  title?: string
    73→}) {
    74→  const handleMouseDown = useCallback(
    75→    (e: React.MouseEvent) => {
    76→      e.preventDefault()
    77→      onClick()
    78→    },
    79→    [onClick],
    80→  )
    81→
    82→  return (
    83→    <button
    84→      type="button"
    85→      title={title}
    86→      disabled={disabled}
    87→      className={cn(
    88→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    89→        'hover:bg-accent hover:text-accent-foreground',
    90→        'active:scale-95',
    91→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    92→        active && 'bg-accent text-accent-foreground'
    93→      )}
    94→      onMouseDown={handleMouseDown}
    95→    >
    96→      {children}
    97→    </button>
    98→  )
    99→}

   306→  const processRecipients = (value: string): string => {
   307→    return value

   317→        // If it contains spaces or looks like a name, return as-is (will fail validation)
   318→        return r
   319→      })
   320→      .join(', ')

   403→  const handleSend = async (scheduledAtDate?: Date | null) => {
   404→    // Prevent double-send from double-click or rapid taps
   405→    if (isSendingRef.current) return
   406→    isSendingRef.current = true

   409→      const e: Record<string, string> = {}

   411→      if (!subject.trim()) e.subject = 'Subject is required'
   412→      const text = editor?.getText().trim()
   413→      if (!text || text.length === 0) e.body = 'Message body is required'
   414→      // Validate CC recipients if provided
   415→      if (showCc && cc.trim()) {
   416→        const ccParts = cc.split(',').map(r => r.trim()).filter(Boolean)
   417→        for (const part of ccParts) {
   418→          const processed = processRecipients(part)
   419→          if (!processed.endsWith('@ezy.af')) {
   420→            e.cc = `"${part}" is not a valid @ezy.af address`
   421→            break
   422→          }
   423→        }
   424→      }
   425→      // Validate BCC recipients if provided
   426→      if (showBcc && bcc.trim()) {
   427→        const bccParts = bcc.split(',').map(r => r.trim()).filter(Boolean)
   428→        for (const part of bccParts) {
   429→          const processed = processRecipients(part)
   430→          if (!processed.endsWith('@ezy.af')) {
   431→            e.bcc = `"${part}" is not a valid @ezy.af address`
   432→            break
   433→          }
   434→        }
   435→      }
   436→      setErrors(e)
   437→      if (Object.keys(e).length > 0) return
   438→
   439→      const html = editor?.getHTML() || ''
   440→      const currentAttachments = attachments
   441→
   442→      // Upload attachments FIRST
   443→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined
   444→      if (currentAttachments.length > 0) {
   445→        setIsUploading(true)
   446→        setUploadProgress(0)
   447→        try {
   448→          const result = await new Promise<Array<{ name: string; url: string; size: number; type: string; data: string }>>((resolve, reject) => {
   449→            const formData = new FormData()
   450→            currentAttachments.forEach(file => formData.append('files', file))
   451→            const xhr = new XMLHttpRequest()
   452→            xhr.open('POST', '/api/upload')
   453→            xhr.withCredentials = true
   454→            xhr.upload.addEventListener('progress', (ev) => {
   455→              if (ev.lengthComputable) {
   456→                setUploadProgress(Math.round((ev.loaded / ev.total) * 100))
   457→              }
   458→            })
   459→            xhr.addEventListener('load', () => {
   460→              if (xhr.status >= 200 && xhr.status < 300) {
   461→                try {
   462→                  const data = JSON.parse(xhr.responseText)
   463→                  resolve(data.files)
   464→                } catch {
   465→                  reject(new Error('Invalid response from server'))
   466→                }
   467→              } else {
   468→                try {
   469→                  const err = JSON.parse(xhr.responseText)
   470→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   471→                } catch {
   472→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   473→                }
   474→              }
   475→            })
   476→            xhr.addEventListener('error', () => reject(new Error('Network error — check your connection')))
   477→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))
   478→            xhr.timeout = 120000 // 2 minute timeout
   479→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))
   480→            xhr.send(formData)
   481→          })
   482→          uploadedFiles = result
   483→        } catch (err) {
   484→          toast.error(err instanceof Error ? err.message : 'Failed to upload attachments')
   485→          setIsUploading(false)
   486→          setUploadProgress(0)
   487→          return
   488→        }
   489→        setIsUploading(false)
   490→        setUploadProgress(0)
   491→      }
   492→
   493→      const data: PendingSendData = {
   494→        to, cc, bcc, subject, html,
   495→        replyToId: replyToEmail?.id,
   496→        attachments: uploadedFiles,
   497→        priority,
   498→        scheduledAt: scheduledAtDate ? scheduledAtDate.toISOString() : null,
   499→      }
   500→      pendingSendRef.current = data
   501→
   502→      // Close compose modal
   503→      setComposeOpen(false)
   504→      setShowSchedulePopover(false)
   505→
   506→      if (scheduledAtDate) {
   507→        // Schedule send — send immediately to API (no undo)
   508→        toast.success(`Email scheduled for ${format(scheduledAtDate, 'MMM d, yyyy h:mm a')}`)
   509→        actuallySendEmail(data)
   510→      } else {
   511→        // Normal send with undo
   512→        const sendResult = await actuallySendEmailWithResponse(data)
   513→        if (sendResult?.emailId) {
   514→          pendingSendRef.current = { ...data, sentEmailId: sendResult.emailId }
   515→        }
   516→
   517→        toast('Message sent.', {
   518→          description: undoCountdown > 0 ? `Undo available (${undoCountdown}s)` : undefined,
   519→          action: {
   520→            label: 'Undo',
   521→            onClick: handleUndo,
   522→          },
   523→          duration: 5000,
   524→        })
   525→
   526→        startUndoCountdown()
   527→
   528→        undoTimerRef.current = setTimeout(() => {
   529→          const payload = pendingSendRef.current
   530→          if (!payload) return
   531→          pendingSendRef.current = null
   532→          if (countdownTimerRef.current) clearInterval(countdownTimerRef.current)
   533→          setUndoCountdown(0)
   534→        }, 5000)
   535→      }
   536→    } finally {
   537→      isSendingRef.current = false
   538→    }
   539→  }

   541→  const actuallySendEmailWithResponse = useCallback(async (data: PendingSendData) => {
   542→    try {
   543→      const finalTo = processRecipients(data.to)
   544→      const payload: Record<string, unknown> = {
   545→        to: finalTo,
   546→        subject: data.subject,
   547→        body: data.html.replace(/<[^>]*>/g, ''),
   548→        bodyHtml: data.html,
   549→      }
   550→      if (data.cc.trim()) payload.cc = processRecipients(data.cc)
   551→      if (data.bcc.trim()) payload.bcc = processRecipients(data.bcc)
   552→      if (data.replyToId) payload.replyToId = data.replyToId
   553→      if (data.attachments && data.attachments.length > 0) {
