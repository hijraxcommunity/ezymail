   434→   434→        bodyHtml: data.html,
   435→   435→      }

   437→   437→      if (data.bcc.trim()) payload.bcc = processRecipients(data.bcc)
   438→   438→      if (data.replyToId) payload.replyToId = data.replyToId
   439→   439→      if (data.attachments && data.attachments.length > 0) {
   440→   440→        payload.attachments = data.attachments.map(a => ({ name: a.name, url: a.url, size: a.size, type: a.type, data: a.data }))
   441→   441→      }

   443→   443→      if (data.scheduledAt) payload.scheduledAt = data.scheduledAt
   444→   444→
   445→   445→      const res = await fetch('/api/emails', {
   446→   446→        method: 'POST',
   447→   447→        headers: { 'Content-Type': 'application/json' },
   448→   448→        body: JSON.stringify(payload),
   449→   449→      })

   452→   452→        const errData = await res.json()
