   272→function DashboardTab() {
   273→  const [stats, setStats] = useState<AdminStats | null>(null)
   274→  const [bStats, setBStats] = useState<BusinessStats | null>(null)
   275→  const [loading, setLoading] = useState(true)
   276→  const [serverTime, setServerTime] = useState(new Date())
   277→
   278→  const fetchStats = useCallback(async () => {
   279→    setLoading(true)
   280→    try {
   281→      const res = await fetch('/api/admin/stats')
   282→      const json = await res.json()
   283→      if (json.success && json.data) {
   284→        setStats(json.data)
   285→      }
   286→    } catch {
   287→      toast.error('Failed to load stats')
   288→    } finally {
   289→      setLoading(false)
   290→    }
   291→  }, [])
   292→
   293→  useEffect(() => {
   294→    fetchStats()
   295→  }, [fetchStats])
   296→
   297→  const fetchBusinessStats = useCallback(async () => {
   298→    try {
   299→      const res = await fetch('/api/admin/business-stats')
   300→      const json = await res.json()
   301→      if (json.success && json.data) {
   302→        setBStats(json.data)
   303→      }
   304→    } catch { /* silent */ }
   305→  }, [])
   306→
   307→  useEffect(() => {
   308→    fetchBusinessStats()
   309→  }, [fetchBusinessStats])
   310→
   311→  useEffect(() => {
   312→    const interval = setInterval(() => setServerTime(new Date()), 1000)
   313→    return () => clearInterval(interval)
   314→  }, [])
   315→
   316→  const emailsChartData = (stats?.emailsPerDay || []).map((d) => ({
   317→    ...d,
   318→    day: new Date(d.date + 'T00:00:00').toLocaleDateString('en', { weekday: 'short' }),
   319→  }))
   320→
   321→  const folderPieData = (stats?.folderDistribution || []).map((d) => ({
   322→    name: d.folder.charAt(0).toUpperCase() + d.folder.slice(1),
   323→    value: d.count,
   324→    color: FOLDER_COLORS[d.folder.toLowerCase()] || '#6B7280',
   325→  }))
   326→
   327→  const usersLineData = (stats?.usersPerDay || []).map((d) => ({
   328→    ...d,
   329→    shortDate: d.date.slice(5), // MM-DD
   330→  }))
   331→
   332→  const statCards = [
   333→    { label: 'Total Users', value: stats?.totalUsers ?? 0, icon: Users, gradient: 'from-blue-500/10 to-blue-600/5', iconBg: 'bg-[#4285F4]' },
   334→    { label: 'Active Now', value: stats?.activeNow ?? 0, icon: Zap, gradient: 'from-green-500/10 to-green-600/5', iconBg: 'bg-[#34A853]' },
   335→    { label: 'Emails Sent', value: stats?.totalEmails ?? 0, icon: Mail, gradient: 'from-yellow-500/10 to-yellow-600/5', iconBg: 'bg-[#FBBC05]' },
   336→    { label: 'Storage Used', value: stats?.storageUsed ?? '0 MB', icon: HardDrive, gradient: 'from-red-500/10 to-red-600/5', iconBg: 'bg-[#EA4335]' },
   337→    { label: 'Pending Reports', value: stats?.pendingReports ?? 0, icon: FileText, gradient: 'from-orange-500/10 to-orange-600/5', iconBg: 'bg-[#F97316]' },
   338→    { label: 'New (7d)', value: stats?.recentRegistrations ?? 0, icon: Activity, gradient: 'from-purple-500/10 to-purple-600/5', iconBg: 'bg-[#9333EA]' },
   339→  ]
   340→
   341→  return (
   342→    <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="space-y-4">
   343→      {/* Stat cards */}
   344→      <div className="grid grid-cols-2 md:grid-cols-3 lg:grid-cols-6 gap-3">
   345→        {statCards.map((card, i) => {
   346→          const Icon = card.icon
   347→          return (
   348→            <motion.div key={card.label} {...fadeInUp} transition={{ delay: i * 0.05 }}>
   349→              <Card className="border-0 shadow-sm relative overflow-hidden">
   350→                <div className={`absolute inset-0 bg-gradient-to-br ${card.gradient} pointer-events-none`} />
   351→                <CardContent className="p-3.5 relative">
   352→                  <div className="flex items-center gap-2.5">
   353→                    <div className={`w-9 h-9 rounded-lg ${card.iconBg} flex items-center justify-center shrink-0`}>
   354→                      <Icon className="w-4 h-4 text-white" />
   355→                    </div>
   356→                    <div className="min-w-0">
   357→                      <p className="text-[11px] text-gray-500 truncate">{card.label}</p>
   358→                      <span className="text-lg font-bold text-[#1F1F1F] dark:text-white leading-tight">
   359→                        {loading ? <Skeleton className="h-5 w-14 inline-block" /> : card.value}
   360→                      </span>
   361→                    </div>
   362→                  </div>
   363→                </CardContent>
   364→              </Card>
   365→            </motion.div>
   366→          )
   367→        })}
   368→      </div>
   369→
   370→      {/* Business Stats */}
   371→      <motion.div {...fadeInUp} transition={{ delay: 0.1 }}>
   372→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   373→          {[
   374→            { label: 'Business Accounts', value: bStats?.totalBusinessAccounts ?? 0, icon: Building2, gradient: 'from-teal-500/10 to-teal-600/5', iconBg: 'bg-teal-600' },
   375→            { label: 'Pending Verifications', value: bStats?.verificationsPendingReview ?? 0, icon: ClipboardCheck, gradient: 'from-yellow-500/10 to-yellow-600/5', iconBg: 'bg-[#FBBC05]' },
   376→            { label: 'Active Subscriptions', value: bStats?.subscriptionBreakdown?.active ?? 0, icon: CheckCircle, gradient: 'from-green-500/10 to-green-600/5', iconBg: 'bg-[#34A853]' },
   377→            { label: 'Trial Accounts', value: bStats?.subscriptionBreakdown?.trial ?? 0, icon: Clock, gradient: 'from-blue-500/10 to-blue-600/5', iconBg: 'bg-[#4285F4]' },
   378→          ].map((card, i) => {
   379→            const Icon = card.icon
   380→            return (
   381→              <Card key={card.label} className="border-0 shadow-sm relative overflow-hidden">
   382→                <div className={`absolute inset-0 bg-gradient-to-br ${card.gradient} pointer-events-none`} />
   383→                <CardContent className="p-3.5 relative">
   384→                  <div className="flex items-center gap-2.5">
   385→                    <div className={`w-9 h-9 rounded-lg ${card.iconBg} flex items-center justify-center shrink-0`}>
   386→                      <Icon className="w-4 h-4 text-white" />
   387→                    </div>
   388→                    <div className="min-w-0">
   389→                      <p className="text-[11px] text-gray-500 truncate">{card.label}</p>
   390→                      <span className="text-lg font-bold text-[#1F1F1F] dark:text-white leading-tight">
   391→                        {loading ? <Skeleton className="h-5 w-14 inline-block" /> : card.value}
   392→                      </span>
   393→                    </div>
   394→                  </div>
   395→                </CardContent>
   396→              </Card>
   397→            )
   398→          })}
   399→        </div>
   400→      </motion.div>
   401→
   402→      {/* Charts row 1 */}
   403→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   404→        {/* Bar chart - Emails per day */}
   405→        <motion.div {...fadeInUp} transition={{ delay: 0.15 }}>
   406→          <Card>
   407→            <CardHeader className="pb-2">
   408→              <div className="flex items-center justify-between">
   409→                <div>
   410→                  <CardTitle className="text-sm font-semibold text-[#1F1F1F] dark:text-white">
   411→                    Emails per Day
   412→                  </CardTitle>
   413→                  <p className="text-xs text-gray-500">Last 7 days</p>
   414→                </div>
   415→                <Button variant="ghost" size="icon" className="h-7 w-7" onClick={fetchStats}>
   416→                  <RefreshCw className={`w-3.5 h-3.5 ${loading ? 'animate-spin' : ''}`} />
   417→                </Button>
   418→              </div>
   419→            </CardHeader>
   420→            <CardContent className="pt-0">
   421→              <div className="h-56">
   422→                {loading ? (
   423→                  <Skeleton className="h-full w-full rounded-lg" />

   437→        </motion.div>
   438→
   439→        {/* Line chart - Users per day */}
   440→        <motion.div {...fadeInUp} transition={{ delay: 0.2 }}>
   441→          <Card>

   443→              <div>
   444→                <CardTitle className="text-sm font-semibold text-[#1F1F1F] dark:text-white">

   447→                <p className="text-xs text-gray-500">Last 30 days</p>
   448→              </div>
   449→            </CardHeader>
   450→            <CardContent className="pt-0">

   454→                ) : (
   455→                  <ResponsiveContainer width="100%" height="100%">
   456→                    <LineChart data={usersLineData}>
   457→                      <CartesianGrid strokeDasharray="3 3" stroke="#E5E7EB" vertical={false} />
   458→                      <XAxis dataKey="shortDate" axisLine={false} tickLine={false} tick={{ fontSize: 10, fill: '#9CA3AF' }} interval={6} />
   459→                      <YAxis axisLine={false} tickLine={false} tick={{ fontSize: 11, fill: '#9CA3AF' }} />
   460→                      <Tooltip contentStyle={CHART_TOOLTIP_STYLE} />

   465→              </div>

   472→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">

   474→        <motion.div {...fadeInUp} transition={{ delay: 0.25 }}>
   475→          <Card className="h-full">
   476→            <CardHeader className="pb-2">
   477→              <CardTitle className="text-sm font-semibold text-[#1F1F1F] dark:text-white">
   478→                Folder Distribution

   480→              <p className="text-xs text-gray-500">Emails by folder</p>
   481→            </CardHeader>
   482→            <CardContent className="pt-0">
   483→              <div className="h-48 flex items-center">
   484→                {loading ? (

   509→                      </ResponsiveContainer>
   510→                    </div>
   511→                    <div className="w-1/2 space-y-2.5 pl-2">
   512→                      {folderPieData.map((item) => (
