    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   103→function ChipInput({ value, onChange, placeholder, onErrorClear }: {
   104→  value: string
   105→  onChange: (val: string) => void
   106→  placeholder?: string
   107→  onErrorClear?: () => void
   108→}) {
   109→  const [inputVal, setInputVal] = useState('')
   110→  const inputRef = useRef<HTMLInputElement>(null)
   111→
   112→  const chips = useMemo(() =>
   113→    value ? value.split(',').map(e => e.trim()).filter(Boolean) : []
   114→  , [value])
   115→
   116→  const commitInput = useCallback((text: string) => {
   117→    const trimmed = text.trim().replace(/,$/g, '')
   118→    if (!trimmed) return
   119→    const current = value ? value.split(',').map(e => e.trim()).filter(Boolean) : []
   120→    if (!current.includes(trimmed)) {
   121→      current.push(trimmed)
   122→      onChange(current.join(', '))
   123→    }
   124→    setInputVal('')
   125→  }, [value, onChange])
   126→
   127→  const removeChip = useCallback((index: number) => {
   128→    const current = value ? value.split(',').map(e => e.trim()).filter(Boolean) : []
   129→    current.splice(index, 1)
   130→    onChange(current.join(', '))
   131→  }, [value, onChange])
   132→
   133→  const handleKeyDown = (e: React.KeyboardEvent) => {
   134→    if (e.key === ',' || e.key === 'Enter') {
   135→      e.preventDefault()
   136→      commitInput(inputVal)
   137→    }
   138→    if (e.key === 'Backspace' && !inputVal && chips.length > 0) {
   139→      removeChip(chips.length - 1)
   140→    }
   141→  }
   142→
   143→  const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   144→    const val = e.target.value
   145→    if (val.includes(',')) {
   146→      const parts = val.split(',')
   147→      parts.slice(0, -1).forEach(p => commitInput(p))
   148→      setInputVal(parts[parts.length - 1])
   149→    } else {
   150→      setInputVal(val)
   151→    }
   152→    onErrorClear?.()
   153→  }
   154→
   155→  return (
   156→    <div
   157→      onClick={() => inputRef.current?.focus()}
   158→      className="flex items-center gap-1.5 flex-wrap flex-1 min-w-0 min-h-[40px] py-1.5 cursor-text"
   159→    >
   160→      {chips.map((chip, i) => (
   161→        <span
   162→          key={`${chip}-${i}`}
   163→          className="inline-flex items-center gap-1 bg-[#D3E3FD] dark:bg-[#4285F4]/20 text-[#1a73e8] dark:text-[#8AB4F8] text-xs font-medium rounded-full px-2 py-0.5 max-w-[200px]"
   164→        >
   165→          <span className="truncate">{chip}</span>
   166→          <button
   167→            type="button"
   168→            onClick={(e) => { e.stopPropagation(); removeChip(i) }}
   169→            className="shrink-0 hover:bg-[#4285F4]/20 rounded-full p-px cursor-pointer"
   170→          >
   171→            <X className="w-3 h-3" />
   172→          </button>
   173→        </span>
   174→      ))}
   175→      <input
   176→        ref={inputRef}
   177→        type="text"
   178→        value={inputVal}
   179→        onChange={handleChange}
   180→        onKeyDown={handleKeyDown}
   181→        placeholder={chips.length === 0 ? placeholder : ''}
   182→        className="flex-1 min-w-[120px] border-0 shadow-none focus-visible:ring-0 text-sm p-0 bg-transparent outline-none h-7"
   183→      />
   184→    </div>
   185→  )
   186→}
