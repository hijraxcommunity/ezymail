   217→function DashboardTab() {
   218→  const [stats, setStats] = useState<AdminStats | null>(null)
   219→  const [loading, setLoading] = useState(true)
   220→  const [serverTime, setServerTime] = useState(new Date())
   221→
   222→  const fetchStats = useCallback(async () => {
   223→    setLoading(true)
   224→    try {
   225→      const res = await fetch('/api/admin/stats')
   226→      const json = await res.json()
   227→      if (json.success && json.data) {
   228→        setStats(json.data)
   229→      }
   230→    } catch {
   231→      toast.error('Failed to load stats')
   232→    } finally {
   233→      setLoading(false)
   234→    }
   235→  }, [])
   236→
   237→  useEffect(() => {
   238→    fetchStats()

   240→
   241→  useEffect(() => {
   242→    const interval = setInterval(() => setServerTime(new Date()), 1000)
   243→    return () => clearInterval(interval)
   244→  }, [])
   245→
   246→  const emailsChartData = (stats?.emailsPerDay || []).map((d) => ({
   247→    ...d,
   248→    day: new Date(d.date + 'T00:00:00').toLocaleDateString('en', { weekday: 'short' }),
   249→  }))
   250→
   251→  const folderPieData = (stats?.folderDistribution || []).map((d) => ({
   252→    name: d.folder.charAt(0).toUpperCase() + d.folder.slice(1),
   253→    value: d.count,
   254→    color: FOLDER_COLORS[d.folder.toLowerCase()] || '#6B7280',

   262→  const statCards = [
   263→    { label: 'Total Users', value: stats?.totalUsers ?? 0, icon: Users, gradient: 'from-blue-500/10 to-blue-600/5', iconBg: 'bg-[#4285F4]' },
   264→    { label: 'Active Now', value: stats?.activeNow ?? 0, icon: Zap, gradient: 'from-green-500/10 to-green-600/5', iconBg: 'bg-[#34A853]' },

   266→    { label: 'Storage Used', value: stats?.storageUsed ?? '0 MB', icon: HardDrive, gradient: 'from-red-500/10 to-red-600/5', iconBg: 'bg-[#EA4335]' },

   275→        {statCards.map((card, i) => {
   276→          const Icon = card.icon
   277→          return (
   278→            <motion.div key={card.label} {...fadeInUp} transition={{ delay: i * 0.05 }}>
   279→              <Card className="border-0 shadow-sm relative overflow-hidden">

   286→                    <div className="min-w-0">
   287→                      <p className="text-[11px] text-gray-500 truncate">{card.label}</p>
   288→                      <span className="text-lg font-bold text-[#1F1F1F] dark:text-white leading-tight">
   289→                        {loading ? <Skeleton className="h-5 w-14 inline-block" /> : card.value}
   290→                      </span>
   291→                    </div>
   292→                  </div>
   293→                </CardContent>
   294→              </Card>
   295→            </motion.div>
   296→          )
   297→        })}
