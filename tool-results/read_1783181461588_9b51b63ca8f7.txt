    62→function ToolbarButton({
    63→  active,
    64→  onClick,
    65→  disabled,
    66→  children,
    67→  title,
    68→}: {
    69→  active?: boolean
    70→  onClick: () => void
    71→  disabled?: boolean
    72→  children: React.ReactNode
    73→  title?: string
    74→}) {
    75→  const handleMouseDown = useCallback(
    76→    (e: React.MouseEvent) => {
    77→      e.preventDefault()
    78→      onClick()
    79→    },
    80→    [onClick],
    81→  )
    82→
    83→  return (
    84→    <button
    85→      type="button"
    86→      title={title}
    87→      disabled={disabled}
    88→      className={cn(
    89→        'inline-flex items-center justify-center h-9 w-9 rounded-lg shrink-0 transition-colors cursor-pointer',
    90→        'hover:bg-accent hover:text-accent-foreground',
    91→        'active:scale-95',
    92→        'disabled:opacity-40 disabled:cursor-not-allowed disabled:pointer-events-none',
    93→        active && 'bg-accent text-accent-foreground'
    94→      )}
    95→      onMouseDown={handleMouseDown}
    96→    >
    97→      {children}
    98→    </button>
    99→  )
   100→}

   441→        }
   442→      }

   451→          }

   457→        return
   458→      }

   461→      const currentAttachments = attachments

   464→      let uploadedFiles: Array<{ name: string; url: string; size: number; type: string }> | undefined

   469→          const result = await new Promise<Array<{ name: string; url: string; size: number; type: string; data: string }>>((resolve, reject) => {
   470→            const formData = new FormData()
   471→            currentAttachments.forEach(file => formData.append('files', file))
   472→            const xhr = new XMLHttpRequest()
   473→            xhr.open('POST', '/api/upload')
   474→            xhr.withCredentials = true
   475→            xhr.upload.addEventListener('progress', (ev) => {
   476→              if (ev.lengthComputable) {
   477→                setUploadProgress(Math.round((ev.loaded / ev.total) * 100))
   478→              }
   479→            })
   480→            xhr.addEventListener('load', () => {
   481→              if (xhr.status >= 200 && xhr.status < 300) {
   482→                try {
   483→                  const data = JSON.parse(xhr.responseText)
   484→                  resolve(data.files)
   485→                } catch {
   486→                  reject(new Error('Invalid response from server'))
   487→                }
   488→              } else {
   489→                try {
   490→                  const err = JSON.parse(xhr.responseText)
   491→                  reject(new Error(err.error || `Upload failed (status ${xhr.status})`))
   492→                } catch {
   493→                  reject(new Error(`Upload failed (status ${xhr.status})`))
   494→                }
   495→              }
   496→            })
   497→            xhr.addEventListener('error', () => reject(new Error('Network error — check your connection')))
   498→            xhr.addEventListener('abort', () => reject(new Error('Upload cancelled')))
   499→            xhr.timeout = 120000 // 2 minute timeout
   500→            xhr.addEventListener('timeout', () => reject(new Error('Upload timed out — try a smaller file')))
   501→            xhr.send(formData)
   502→          })
   503→          uploadedFiles = result
   504→        } catch (err) {
   505→          toast.error(err instanceof Error ? err.message : 'Failed to upload attachments')
   506→          setIsUploading(false)
   507→          setUploadProgress(0)
   508→          return
   509→        }
   510→        setIsUploading(false)
   511→        setUploadProgress(0)
   512→      }
   513→
   514→      const data: PendingSendData = {
   515→        to, cc, bcc, subject, html,
   516→        replyToId: replyToEmail?.id,
   517→        attachments: uploadedFiles,
   518→        priority,
   519→        scheduledAt: scheduledAtDate ? scheduledAtDate.toISOString() : null,
   520→      }
   521→      pendingSendRef.current = data
   522→
   523→      // Close compose modal
   524→      setComposeOpen(false)
   525→      setShowSchedulePopover(false)
   526→
   527→      if (scheduledAtDate) {
   528→        // Schedule send — send immediately to API (no undo)
   529→        toast.success(`Email scheduled for ${format(scheduledAtDate, 'MMM d, yyyy h:mm a')}`)
   530→        actuallySendEmail(data)
   531→        // Delete draft if editing one
   532→        if (editDraftEmail) {
   533→          fetch(`/api/emails/${editDraftEmail.id}`, { method: 'DELETE' }).catch(() => {})
   534→        }
   535→      } else {
   536→        // Normal send with undo
   537→        const sendResult = await actuallySendEmailWithResponse(data)
   538→        if (sendResult?.emailId) {
   539→          pendingSendRef.current = { ...data, sentEmailId: sendResult.emailId }
   540→          // Delete draft if editing one
   541→          if (editDraftEmail) {
   542→            fetch(`/api/emails/${editDraftEmail.id}`, { method: 'DELETE' }).catch(() => {})
   543→          }
   544→        }
   545→
   546→        toast('Message sent.', {
   547→          description: undoCountdown > 0 ? `Undo available (${undoCountdown}s)` : undefined,
   548→          action: {
   549→            label: 'Undo',
   550→            onClick: handleUndo,
   551→          },
   552→          duration: 5000,
